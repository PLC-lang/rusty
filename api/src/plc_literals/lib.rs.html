1//! Allocation-free parsers for the Structured Text literal grammar.
2//!
3//! The compiler's lexer has already classified a token when these functions run; the standard
4//! library's `STRING_TO_*` functions call them on raw runtime input. Both sides share one
5//! implementation, so a literal that compiles to a value converts to the same value at runtime.
6//!
7//! Nothing here allocates or touches floating point: every result is exact.
8#![no_std]
9
10pub mod calendar;
11pub mod duration;
12pub mod integer;

20/// The whitespace Structured Text treats as insignificant: space, tab, CR, LF, VT, FF.
21pub fn is_whitespace(c: char) -> bool {
22    matches!(c, ' ' | '\t' | '\r' | '\n' | '\x0B' | '\x0C')
23}
24
25/// Trims [`is_whitespace`] characters from both ends of `input`.
26pub fn trim(input: &str) -> &str {
27    input.trim_matches(is_whitespace)
28}
29
30/// Strips the first of `prefixes` that `input` starts with (ASCII case-insensitively) and returns
31/// the remainder, or `None` if no prefix matches.
32pub fn strip_prefix_ignore_ascii_case<'a>(input: &'a str, prefixes: &[&str]) -> Option<&'a str> {
33    prefixes.iter().find_map(|prefix| {
34        let head = input.get(..prefix.len())?;
35        head.eq_ignore_ascii_case(prefix).then_some(&input[prefix.len()..])
36    })
37}

39/// Splits off the longest leading digit group `digit+ ('_' digit+)*` for `radix` and returns
40/// `(group, rest)`. `group` is empty when `input` does not start with a digit; a trailing `_`
41/// is never part of the group.
42pub(crate) fn split_digit_group(input: &str, radix: u32) -> (&str, &str) {
43    let bytes = input.as_bytes();
44    let mut end = 0;
45    let mut index = 0;
46    while index < bytes.len() {
47        let byte = bytes[index];
48        if (byte as char).is_digit(radix) {
49            index += 1;
50            end = index;
51        } else if byte == b'_' && end == index && end > 0 {
52            // an underscore may follow a digit, but only counts once the next digit arrives
53            index += 1;
54        } else {
55            break;
56        }
57    }
58    input.split_at(end)
59}
60
61/// Iterates the digit values of a group produced by [`split_digit_group`], skipping underscores.
62pub(crate) fn digit_values(group: &str, radix: u32) -> impl DoubleEndedIterator<Item = u32> + Clone + '_ {
63    group.chars().filter(|c| *c != '_').map(move |c| c.to_digit(radix).unwrap_or(0))
64}
65
66/// Accumulates a digit group into a `u128`, or `None` on overflow.
67pub(crate) fn digit_group_value(group: &str, radix: u32) -> Option<u128> {
68    digit_values(group, radix)
69        .try_fold(0u128, |acc, digit| acc.checked_mul(radix as u128)?.checked_add(digit as u128))
70}

72#[cfg(test)]
73mod tests {
74    use super::*;
75
76    #[test]
77    fn digit_groups_stop_at_misplaced_underscores() {
78        assert_eq!(split_digit_group("1_000ms", 10), ("1_000", "ms"));
79        assert_eq!(split_digit_group("1__0", 10), ("1", "__0"));
80        assert_eq!(split_digit_group("1_", 10), ("1", "_"));
81        assert_eq!(split_digit_group("_1", 10), ("", "_1"));
82        assert_eq!(split_digit_group("F_F", 16), ("F_F", ""));
83        assert_eq!(split_digit_group("19", 8), ("1", "9"));
84        assert_eq!(split_digit_group("", 10), ("", ""));
85    }
86
87    #[test]
88    fn prefixes_match_case_insensitively() {
89        assert_eq!(strip_prefix_ignore_ascii_case("time#1s", &["TIME#", "T#"]), Some("1s"));
90        assert_eq!(strip_prefix_ignore_ascii_case("T#1s", &["TIME#", "T#"]), Some("1s"));
91        assert_eq!(strip_prefix_ignore_ascii_case("LT#1s", &["TIME#", "T#"]), None);
92        assert_eq!(strip_prefix_ignore_ascii_case("T", &["TIME#", "T#"]), None);
93        assert_eq!(strip_prefix_ignore_ascii_case("§#", &["T#"]), None);
94    }
95
96    #[test]
97    fn trim_covers_the_structured_text_whitespace_set() {
98        assert_eq!(trim(" \t\r\n\x0B\x0Cx\x0C\x0B\n\r\t "), "x");
99        assert_eq!(trim("\u{A0}x"), "\u{A0}x");
100    }
