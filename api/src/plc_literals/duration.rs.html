1//! Duration literals (`TIME`, `LTIME`): the body after the `#`.

5/// A parsed duration. `nanos` is the magnitude; the sign is kept separately so that callers can
6/// decide what a negative duration means for their target type.
7#[derive(Debug, Clone, Copy, PartialEq, Eq)]
8pub struct Duration {
9    pub negative: bool,
10    pub nanos: u128,
11}
12
13impl Duration {
14    /// The duration as signed nanoseconds, or `None` if the magnitude exceeds `i64`.
15    pub fn signed_nanos(&self) -> Option<i64> {
16        let nanos = i64::try_from(self.nanos).ok()?;
17        Some(if self.negative { -nanos } else { nanos })
18    }
19}
20
21/// The segment units of a duration literal, largest first.
22#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
23pub enum Unit {
24    Day,
25    Hour,
26    Minute,
27    Second,
28    Millisecond,
29    Microsecond,
30    Nanosecond,
31}
32
33impl Unit {
34    /// The unit expressed in nanoseconds as `mantissa * 10^power`.
35    fn factor(self) -> (u128, u32) {
36        match self {
37            Unit::Day => (864, 11),
38            Unit::Hour => (36, 11),
39            Unit::Minute => (6, 10),
40            Unit::Second => (1, 9),
41            Unit::Millisecond => (1, 6),
42            Unit::Microsecond => (1, 3),
43            Unit::Nanosecond => (1, 0),
44        }
45    }
46
47    fn from_symbol(symbol: &str) -> Option<Unit> {
48        [
49            ("d", Unit::Day),
50            ("h", Unit::Hour),
51            ("m", Unit::Minute),
52            ("s", Unit::Second),
53            ("ms", Unit::Millisecond),
54            ("us", Unit::Microsecond),
55            ("ns", Unit::Nanosecond),
56        ]
57        .into_iter()
58        .find_map(|(name, unit)| symbol.eq_ignore_ascii_case(name).then_some(unit))
59    }
60}

62#[derive(Debug, Clone, Copy, PartialEq, Eq)]
63pub enum DurationError {
64    /// No segment at all (`T#`, `T#-`).
65    Empty,
66    /// A decimal point without digits in front of or after it.
67    MissingDigits,
68    /// Digits that are not followed by a unit.
69    MissingUnit,
70    /// Letters that are not one of `d h m s ms us ns`; `start..end` is their byte range in the body.
71    UnknownUnit { start: usize, end: usize },
72    /// A unit larger than the previous one (strict mode only).
73    SegmentOutOfOrder,
74    /// A unit that already appeared (strict mode only).
75    DuplicateSegment,
76    /// Anything else, including whitespace or `_` where the leniency does not allow it.
77    InvalidCharacter,
78    /// More than `u128::MAX` nanoseconds.
79    Overflow,
80}
81
82/// What the parser accepts beyond the literal grammar of the language.
83#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
84pub struct Leniency {
85    /// Whitespace and `_` may separate segments (they never split a number or a unit).
86    pub separators: bool,
87    /// Segments may repeat and appear in any order; their values are summed.
88    pub unordered_segments: bool,
89}
90
91impl Leniency {
92    /// Exactly the literal grammar: `d h m s ms us ns`, each at most once, in that order.
93    pub const STRICT: Leniency = Leniency { separators: false, unordered_segments: false };
94    /// Everything `STRICT` accepts, plus separated, repeated and unordered segments.
95    pub const RUNTIME: Leniency = Leniency { separators: true, unordered_segments: true };
96}

98/// Parses the body of a duration literal: `[+-]? (digits [. digits] unit)+`, where digit groups
99/// may contain single underscores and units are matched case-insensitively. Fractions are exact and
100/// truncated below one nanosecond.
101pub fn parse_duration(body: &str, leniency: Leniency) -> Result<Duration, DurationError> {
102    let (negative, mut rest) = match body.as_bytes().first() {
103        Some(b'-') => (true, &body[1..]),
104        Some(b'+') => (false, &body[1..]),
105        _ => (false, body),
106    };
107    let mut nanos: u128 = 0;
108    let mut previous: Option<Unit> = None;
109
110    loop {
111        if leniency.separators && previous.is_some() {
112            let trimmed = rest.trim_start_matches(|c: char| is_whitespace(c) || c == '_');
113            if trimmed.is_empty() && !rest.is_empty() {
114                return Err(DurationError::InvalidCharacter);
115            }
116            rest = trimmed;
117        }
118        if rest.is_empty() {
119            break;
120        }
121
122        let (integer, after_integer) = split_digit_group(rest, 10);
123        if integer.is_empty() {
124            return Err(match after_integer.as_bytes()[0] {
125                b'.' => DurationError::MissingDigits,
126                _ => DurationError::InvalidCharacter,
127            });
128        }
129        let (fraction, after_number) = match after_integer.strip_prefix('.') {
130            Some(after_dot) => {
131                let (fraction, after_fraction) = split_digit_group(after_dot, 10);
132                if fraction.is_empty() {
133                    return Err(DurationError::MissingDigits);
134                }
135                (fraction, after_fraction)
136            }
137            None => ("", after_integer),
138        };
139
140        let unit_len = after_number.bytes().take_while(u8::is_ascii_alphabetic).count();
141        if unit_len == 0 {
142            return Err(DurationError::MissingUnit);
143        }
144        let start = body.len() - after_number.len();
145        let unit = Unit::from_symbol(&after_number[..unit_len])
146            .ok_or(DurationError::UnknownUnit { start, end: start + unit_len })?;
147        rest = &after_number[unit_len..];
148
149        if let Some(previous) = previous.filter(|_| !leniency.unordered_segments) {
150            if unit == previous {
151                return Err(DurationError::DuplicateSegment);
152            }
153            if unit < previous {
154                return Err(DurationError::SegmentOutOfOrder);
155            }
156        }
157        previous = Some(unit);
158
159        nanos = nanos.checked_add(segment_nanos(integer, fraction, unit)?).ok_or(DurationError::Overflow)?;
160    }
161
162    if previous.is_none() {
163        return Err(DurationError::Empty);
164    }
165    Ok(Duration { negative, nanos })
166}

168/// `integer.fraction` units in nanoseconds, truncated. With the unit being `mantissa * 10^power`
169/// nanoseconds, the decimal point moves `power` places right; whatever fraction digits remain are
170/// multiplied by `mantissa` with schoolbook carries from the right, which yields the exact floor.
171fn segment_nanos(integer: &str, fraction: &str, unit: Unit) -> Result<u128, DurationError> {
172    let (mantissa, power) = unit.factor();
173    let mut fraction_digits = digit_values(fraction, 10);
174    let shifted = (0..power).map(|_| fraction_digits.next().unwrap_or(0));
175    let mut whole: u128 = 0;
176    for digit in digit_values(integer, 10).chain(shifted) {
177        whole = whole
178            .checked_mul(10)
179            .and_then(|it| it.checked_add(digit as u128))
180            .ok_or(DurationError::Overflow)?;
181    }
182    let carry = fraction_digits.rev().fold(0u128, |carry, digit| (mantissa * digit as u128 + carry) / 10);
183    whole.checked_mul(mantissa).and_then(|it| it.checked_add(carry)).ok_or(DurationError::Overflow)
184}

186#[cfg(test)]
187mod tests {
188    use super::*;

190    const NANOS_PER_MILLI: u128 = 1_000_000;
191    const NANOS_PER_SECOND: u128 = 1_000 * NANOS_PER_MILLI;
192    const NANOS_PER_MINUTE: u128 = 60 * NANOS_PER_SECOND;
193    const NANOS_PER_HOUR: u128 = 60 * NANOS_PER_MINUTE;
194    const NANOS_PER_DAY: u128 = 24 * NANOS_PER_HOUR;
195
196    fn strict(body: &str) -> Result<Duration, DurationError> {
197        parse_duration(body, Leniency::STRICT)
198    }
199    fn runtime(body: &str) -> Result<Duration, DurationError> {
200        parse_duration(body, Leniency::RUNTIME)
201    }
202    fn positive(nanos: u128) -> Result<Duration, DurationError> {
203        Ok(Duration { negative: false, nanos })
204    }
205
206    #[test]
207    fn every_unit() {
208        assert_eq!(strict("1d"), positive(NANOS_PER_DAY));
209        assert_eq!(strict("1h"), positive(NANOS_PER_HOUR));
210        assert_eq!(strict("1m"), positive(NANOS_PER_MINUTE));
211        assert_eq!(strict("1s"), positive(NANOS_PER_SECOND));
212        assert_eq!(strict("1ms"), positive(NANOS_PER_MILLI));
213        assert_eq!(strict("1us"), positive(1_000));
214        assert_eq!(strict("1ns"), positive(1));
215        assert_eq!(strict("1D2H3M4S5MS6US7NS"), strict("1d2h3m4s5ms6us7ns"));
216        assert_eq!(
217            strict("1d2h3m4s5ms6us7ns"),
218            positive(
219                NANOS_PER_DAY + 2 * NANOS_PER_HOUR + 3 * NANOS_PER_MINUTE + 4 * NANOS_PER_SECOND + 5_006_007
220            )
221        );
222    }

224    #[test]
225    fn signs() {
226        assert_eq!(strict("-1s"), Ok(Duration { negative: true, nanos: NANOS_PER_SECOND }));
227        assert_eq!(strict("+1s"), positive(NANOS_PER_SECOND));
228        assert_eq!(strict("-1s").unwrap().signed_nanos(), Some(-(NANOS_PER_SECOND as i64)));
229        assert_eq!(strict("+-1s"), Err(DurationError::InvalidCharacter));
230        assert_eq!(strict("1s+1s"), Err(DurationError::InvalidCharacter));
231        assert_eq!(strict("-"), Err(DurationError::Empty));
232    }

243        assert_eq!(strict("1.5ns"), positive(1));
244        assert_eq!(strict("0.1us"), positive(100));
245        assert_eq!(strict("2.75s"), positive(2_750 * NANOS_PER_MILLI));
246        assert_eq!(strict("1.s"), Err(DurationError::MissingDigits));

248        assert_eq!(strict("1,5s"), Err(DurationError::MissingUnit));
