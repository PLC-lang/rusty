1//! The serde model of the `.cfc` document, the shape of the XML format itself.
2
3// quick-xml's serde support strips namespace prefixes and matches on the local
4// name, so the `rename`s use bare names (`Function`, `type`) rather than their
5// `ppx:` / `xsi:` prefixed forms, and attributes take a leading `@`.
6
7use serde::Deserialize;
8
9#[derive(Debug, Deserialize)]
10pub enum Pou {
11    #[serde(rename = "Function")]
12    Function(PouContent),
13
14    #[serde(rename = "FunctionBlock")]
15    FunctionBlock(PouContent),
16
17    #[serde(rename = "Program")]
18    Program(PouContent),
19}

22pub enum PouKind {

93// The storage behavior on a data sink. Set/Reset latch: the incoming value no
94// longer assigns but guards a constant `TRUE` (Set) or `FALSE` (Reset) store.
95// Reference re-points: the sink stores the source's address (`REF=`).
96#[derive(Debug, Deserialize)]

100}

232impl Pou {
233    pub fn parse(content: &str) -> Result<Self, quick_xml::DeError> {
234        quick_xml::de::from_str(content)
235    }
236
237    pub fn kind(&self) -> PouKind {
238        match self {
239            Pou::Function(_) => PouKind::Function,
240            Pou::FunctionBlock(_) => PouKind::FunctionBlock,
241            Pou::Program(_) => PouKind::Program,
242        }
243    }
244
245    pub fn content(&self) -> &PouContent {
246        match self {
247            Pou::Function(content) | Pou::FunctionBlock(content) | Pou::Program(content) => content,
248        }
249    }
250}

299    pub fn in_negated(&self) -> bool {
300        self.data(|data| data.negation.as_ref()).is_some_and(|negation| negation.in_negated)
301    }
302
303    pub fn out_negated(&self) -> bool {
304        self.data(|data| data.negation.as_ref()).is_some_and(|negation| negation.out_negated)
305    }
306
307    pub fn priority(&self) -> Option<usize> {
308        self.data(|data| data.evaluation_priority.as_ref()).map(|priority| priority.priority)
309    }
310
311    pub fn storage(&self) -> Option<Storage> {
312        self.data(|data| data.storage_mode.as_ref()).map(|storage| storage.mode)
313    }
314
315    pub fn has_execution_control(&self) -> bool {
316        self.data(|data| data.execution_control.as_ref()).is_some_and(|control| control.enabled)
317    }
318
319    pub fn en_pin(&self) -> Option<&Pin> {
320        self.has_execution_control().then(|| control_pin(self.input_pins(), "EN")).flatten()
321    }
322
323    pub fn eno_pin(&self) -> Option<&Pin> {
324        self.has_execution_control().then(|| control_pin(self.output_pins(), "ENO")).flatten()
325    }
326
327    // `AddData` interleaves its payloads across several `Data` entries.
328    fn data<'a, T>(&'a self, pick: impl Fn(&'a Data) -> Option<&'a T>) -> Option<&'a T> {
329        self.add_data.as_ref()?.data.iter().find_map(pick)
330    }

350// The execution control pin among a block's pins. A callee declaring an own
351// `EN`/`ENO` parameter is indistinguishable from the control pin in the
352// current export format; panic until the format defines the disambiguation.
353fn control_pin<'model>(pins: &'model [Pin], name: &str) -> Option<&'model Pin> {
354    let mut candidates = pins.iter().filter(|pin| pin.parameter_name == name);
355    let pin = candidates.next();
356
357    assert!(
358        candidates.next().is_none(),
359        "ambiguous `{name}` pin: the callee declares an own `{name}` parameter next to the execution control pin"
360    );
361
362    pin
