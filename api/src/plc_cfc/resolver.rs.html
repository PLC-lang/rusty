25enum Role {

27    Sink,

30    Label,

38// A plain variable/literal, or a block output (read as `instance.member`);
39// `negated` carries the inversion bubble of any EN pin hopped on the way.
40enum Source<'model> {
41    Variable { object: &'model FbdObject, negated: bool },
42    Output { block: &'model FbdObject, pin: &'model Pin, negated: bool },
43}
44
45// Everything the linking pass needs to know up front.
46struct Survey<'model> {
47    /// Every `<ConnectionPointOut connectionPointOutId="4">` to its owning element
48    by_pin: HashMap<usize, &'model FbdObject>,
49
50    /// Block `<OutputVariable parameterName="out">` pins by out-id
51    block_output: HashMap<usize, &'model Pin>,
52
53    /// The connector owning each `<Connector label="x">`
54    connector_by_label: HashMap<&'model str, &'model FbdObject>,
55
56    /// Every `<CfcLabel label="top">` name
57    labels: HashSet<&'model str>,
58
59    /// Every `<CfcJump targetLabel="top">`
60    targets: HashSet<&'model str>,
61}
62
63enum Trace<'model> {
64    Reached(Source<'model>),
65    DeadEnd(Broken<'model>),
66    Unwired,
67}
68
69struct Broken<'model> {
70    element: &'model FbdObject,
71    reason: Reason,
72}
73
74enum Reason {
75    OpenConnector,
76    DanglingContinuation,
77    EnoCycle,
78}

85    pub fn resolve(mut self, network: &model::Network) -> (Network, Vec<Diagnostic>) {
86        // Chart the wiring up front; `consumed` decides which outputs need temporaries.
87        let survey = self.survey(network);
88        let consumed = consumed(network, &survey);
89
90        // Turn each element into its statement(s), tagged with its priority.
91        let mut statements = Vec::new();
92        let mut temporaries = Vec::new();
93        let mut broken = Vec::new();
94        for object in network.elements() {
95            let location = self.factory.create_block_location(object.global_id);
96
97            match Role::from(object) {
98                Role::Sink => match trace(consumes(object), &survey) {
99                    // An address has no inverse; a bubble anywhere on a
100                    // reference assignment's wire is rejected.
101                    Trace::Reached(source)
102                        if matches!(object.storage(), Some(Storage::Reference))
103                            && negated(object, &source) =>
104                    {
105                        let name = object.identifier().unwrap_or_default();
106                        self.diagnostics.push(Diagnostic::negated_reference_assignment(name, location));
107                    }
108
109                    // The traced source becomes the sink's assigned value,
110                    // negated once more by the sink's own inversion bubble.
111                    Trace::Reached(source) => {
112                        let sink = self.expression(object, &location);
113                        let source = self.value(&source, &location);
114                        let source = self.negate_if(source, &location, object.in_negated());
115
116                        let statement = Statement::Assignment { sink, source, storage: object.storage() };
117                        statements.push((object.priority(), statement));
118                    }
119
120                    // A consumed chain that never reached a source; reported below.
121                    Trace::DeadEnd(at) => broken.push(at),
122
123                    // An unwired sink assigns nothing; drop it.
124                    Trace::Unwired => {}
125                },
126
127                Role::Return => match trace(consumes(object), &survey) {
128                    // The traced source guards the return.
129                    Trace::Reached(source) => {
130                        let condition = self.condition(object, &source, &location);
131
132                        let statement = Statement::Return { condition, location };
133                        statements.push((object.priority(), statement));
134                    }
135
136                    // A consumed chain that never reached a source; reported below.
137                    Trace::DeadEnd(at) => broken.push(at),
138
139                    // A return without a condition is rejected; drop it.
140                    Trace::Unwired => {
141                        self.diagnostics.push(Diagnostic::disconnected_return(location));
142                    }
143                },
144
145                Role::Jump => {
146                    let target = object.target_label().unwrap_or_default().to_string();
147
148                    // A jump to a name no label defines has nowhere to land.
149                    if !survey.labels.contains(target.as_str()) {
150                        self.diagnostics.push(Diagnostic::undefined_jump_target(&target, location.clone()));
151                    }
152
153                    match trace(consumes(object), &survey) {
154                        // The traced source guards the jump.
155                        Trace::Reached(source) => {
156                            let condition = Some(self.condition(object, &source, &location));
157
158                            let statement = Statement::Jump { condition, target, location };
159                            statements.push((object.priority(), statement));
160                        }
161
162                        // A consumed chain that never reached a source; reported below.
163                        Trace::DeadEnd(at) => broken.push(at),
164
165                        // An unwired jump is legal but can never be taken; warn and keep it.
166                        Trace::Unwired => {
167                            self.diagnostics.push(Diagnostic::disconnected_jump(location.clone()));
168
169                            let statement = Statement::Jump { condition: None, target, location };
170                            statements.push((object.priority(), statement));
171                        }
172                    }
173                }
174
175                Role::Label => {
176                    let name = object.label().unwrap_or_default().to_string();
177
178                    // A label no jump targets is dead routing; keep it, but flag it.
179                    if !survey.targets.contains(name.as_str()) {
180                        self.diagnostics.push(Diagnostic::unused_label(&name, location.clone()));
181                    }
182
183                    let statement = Statement::Label { name, location };
184                    statements.push((object.priority(), statement));
185                }
186
187                // A callee the index doesn't know can't be classified; reject it.
188                Role::Block if block::is_unknown(object, self.index) => {
189                    let name = object.type_name().unwrap_or_default();
190
191                    self.diagnostics.push(Diagnostic::unknown_block_type(name, location));
192                }
193
194                // Stateless blocks (functions and sort of methods); temporaries for outputs
195                Role::Block if block::is_stateless(object, self.index) => {
196                    let variadic = block::is_variadic(object, self.index);
197                    let enable = self.enable(object, &survey, &mut broken, &location);
198
199                    let mut arguments = Vec::new();
200                    for pin in block::inputs(object) {
201                        let argument = match trace(pin.source_pin(), &survey) {
202                            // A variadic callee takes its values by pin order; the pins
203                            // carry no usable parameter names.
204                            Trace::Reached(source) if variadic => {
205                                Some(self.variadic(pin, &source, &location))
206                            }
207
208                            // The traced source becomes the pin's passed value.
209                            Trace::Reached(source) => Some(self.input(pin, &source, &location)),
210
211                            // A consumed chain that never reached a source; reported below.
212                            Trace::DeadEnd(at) => {
213                                broken.push(at);
214                                (!variadic).then(|| self.empty_input(pin, &location))
215                            }
216
217                            // An unwired pin takes the callee's declared default; a
218                            // variadic slot has none and is dropped to keep the
219                            // remaining positions meaningful.
220                            Trace::Unwired => (!variadic).then(|| self.empty_input(pin, &location)),
221                        };
222
223                        arguments.extend(argument);
224                    }
225
226                    // Consumed outputs are captured into temporaries
227                    let mut capture = None;
228                    for pin in object.output_pins() {
229                        // ENO is not a data output; consumers read the EN source.
230                        if object.eno_pin().is_some_and(|eno| std::ptr::eq(eno, pin)) {
231                            continue;
232                        }
233
234                        let read = pin.output_pin().is_some_and(|id| consumed.contains(&id));
235                        let name = read.then(|| block::temp_name(object, pin));
236
237                        if let Some(name) = &name {
238                            if let Some(temporary) = self.temporary(name, object, pin, &location) {
239                                temporaries.push((object.priority(), temporary));
240                            }
241                        }
242
243                        match block::is_return_pin(pin) {
244                            // The return value is received by assigning the call itself.
245                            true => capture = name,
246
247                            // Any other output is received inside the parentheses.
248                            false => arguments.push(Argument::Output {
249                                parameter: pin.parameter_name.clone(),
250                                capture: name,
251                            }),
252                        }
253                    }
254
255                    let statement = Statement::Call {
256                        target: block::call_target(object),
257                        arguments,
258                        capture,
259                        enable,
260                        location,
261                    };
262                    statements.push((object.priority(), statement));
263                }
264
265                // A stateful call passes only its wired inputs; outputs read back as members.
266                Role::Block => {
267                    let enable = self.enable(object, &survey, &mut broken, &location);
268
269                    let mut arguments = Vec::new();
270                    for pin in block::inputs(object) {
271                        match trace(pin.source_pin(), &survey) {
272                            Trace::Reached(source) => arguments.push(self.input(pin, &source, &location)),
273
274                            // A consumed chain that never reached a source; reported below.
275                            Trace::DeadEnd(at) => broken.push(at),
276
277                            // An unwired input passes nothing; the callee keeps last cycle's value.
278                            Trace::Unwired => {}
279                        }
280                    }
281
282                    let statement = Statement::Call {
283                        target: block::call_target(object),
284                        arguments,
285                        capture: None,
286                        enable,
287                        location,
288                    };
289                    statements.push((object.priority(), statement));
290                }
291
292                // Placed but never wired; warn and ignore.
293                Role::Unconnected => {
294                    let name = object.identifier().unwrap_or("<unnamed>");
295
296                    self.diagnostics.push(Diagnostic::unconnected_element(name, location));
297                }
298
299                // Read-only (sources) or pure routing; no statements.
300                Role::Source | Role::Connector | Role::Continuation | Role::Other => {}
301            }
302        }
303
304        // Report a broken element once, no matter how many consumers hit it.
305        broken.sort_by_key(|at| at.element.global_id);
306        broken.dedup_by_key(|at| at.element.global_id);
307        for at in broken {
308            self.diagnostics.push(at.diagnostic(&self.factory));
309        }
310
311        // Statements and temporaries run in evaluation-priority order.
312        statements.sort_by_key(|(priority, _)| priority.unwrap_or(usize::MAX));
313        let statements = statements.into_iter().map(|(_, statement)| statement).collect();
314        temporaries.sort_by_key(|(priority, _)| priority.unwrap_or(usize::MAX));
315        let temporaries = temporaries.into_iter().map(|(_, temporary)| temporary).collect();
316
317        (Network { statements, temporaries }, self.diagnostics)
318    }

320    // `parameter := value`, negated by the pin's inversion bubble.
321    fn input(&mut self, pin: &Pin, source: &Source, location: &SourceLocation) -> Argument {
322        let value = self.value(source, location);
323        let value = self.negate_if(value, location, pin.negated);
324
325        Argument::Input { parameter: pin.parameter_name.clone(), value: Box::new(value) }
326    }

328    // `parameter := `; an empty argument for an unwired parameter.
329    fn empty_input(&mut self, pin: &Pin, location: &SourceLocation) -> Argument {
330        let value = AstFactory::create_empty_statement(location.clone(), self.ids.next_id());
331        Argument::Input { parameter: pin.parameter_name.clone(), value: Box::new(value) }
332    }

336        let value = self.value(source, location);
337        let value = self.negate_if(value, location, pin.negated);

344        name: &str,

346        pin: &Pin,

362    fn survey<'model>(&mut self, network: &'model model::Network) -> Survey<'model> {
363        let mut by_pin = HashMap::new();
364        let mut block_output = HashMap::new();
365        let mut connector_by_label = HashMap::new();
366        let mut labels = HashSet::new();
367        let mut targets = HashSet::new();
368
369        for object in network.elements() {
370            // Register every output pin an incoming wire could reference.
371            if let Some(out) = &object.connection_out {
372                by_pin.insert(out.id, object);
373            }
374
375            match Role::from(object) {
376                // A label name claimed twice makes its jump target ambiguous.
377                Role::Label => {
378                    if let Some(label) = object.label() {
379                        if !labels.insert(label) {
380                            let location = self.factory.create_block_location(object.global_id);
381                            self.diagnostics.push(Diagnostic::duplicate_label(label, location));
382                        }
383                    }
384                }
385
386                Role::Jump => {
387                    if let Some(target) = object.target_label() {
388                        targets.insert(target);
389                    }
390                }
391
392                // A block exposes one pin per output parameter; only the return pin is
393                // unnamed, and the ENO pin resolves to the EN source instead of a member read.
394                Role::Block => {
395                    if object.output_pins().iter().filter(|pin| block::is_return_pin(pin)).count() > 1 {
396                        let name = object.type_name().unwrap_or_default();
397                        let location = self.factory.create_block_location(object.global_id);
398                        self.diagnostics.push(Diagnostic::duplicate_return_pin(name, location));
399                    }
400
401                    let eno = object.eno_pin().and_then(|pin| pin.output_pin());
402                    for pin in object.output_pins() {
403                        if let Some(id) = pin.output_pin() {
404                            by_pin.insert(id, object);
405
406                            if Some(id) != eno {
407                                block_output.insert(id, pin);
408                            }
409                        }
410                    }
411                }
412
413                Role::Connector => {
414                    if let Some(label) = object.label() {
415                        match connector_by_label.entry(label) {
416                            // The first connector to claim a label owns it.
417                            Entry::Vacant(entry) => {
418                                entry.insert(object);
419                            }
420
421                            // A later one reusing the label is a duplicate.
422                            Entry::Occupied(_) => {
423                                let location = self.factory.create_block_location(object.global_id);
424                                self.diagnostics.push(Diagnostic::duplicate_connector(label, location));
425                            }
426                        }
427                    }
428                }
429
430                _ => {}
431            }
432        }
433
434        Survey { by_pin, block_output, connector_by_label, labels, targets }
435    }

437    fn expression(&mut self, object: &FbdObject, location: &SourceLocation) -> AstNode {
438        let node = self.parse(object, location);
439
440        // A variable element may only hold a literal or a reference; anything
441        // else (a call, arithmetic) must be modeled as a block element instead.
442        if !is_supported(&node) {
443            let location = self.factory.create_block_location(object.global_id);
444            let text = object.identifier().unwrap_or_default();
445            self.diagnostics.push(Diagnostic::unsupported_cfc_expression(text, location));
446        }
447
448        node
449    }

451    // The value a source produces: a vetted plain expression, or a block-output
452    // member read; either negated by the producer's inversion bubble, then by
453    // the bubbles of the EN pins the trace hopped through.
454    fn value(&mut self, source: &Source, location: &SourceLocation) -> AstNode {
455        match source {
456            Source::Variable { object, negated } => {
457                let node = self.expression(object, location);
458                let node = self.negate_if(node, location, object.out_negated());
459                self.negate_if(node, location, *negated)
460            }
461            Source::Output { block, pin, negated } => {
462                let node = self.member_read(block, pin, location);
463                self.negate_if(node, location, *negated)
464            }
465        }
466    }

468    // Creates an AST node, wrapped in a NOT expression for each inversion
469    // bubble on the wire: the producer's, then the consumer's on top.
470    fn condition(&mut self, consumer: &FbdObject, source: &Source, location: &SourceLocation) -> AstNode {
471        let node = match source {
472            // Unvetted; a condition may be any ST expression.
473            Source::Variable { object, negated } => {
474                let node = self.parse(object, location);
475                let node = self.negate_if(node, location, object.out_negated());
476                self.negate_if(node, location, *negated)
477            }
478            Source::Output { .. } => self.value(source, location),
479        };
480
481        self.negate_if(node, location, consumer.in_negated())
482    }

484    // The EN guard for a block with execution control, negated by the pin's
485    // inversion bubble; `None` without the flag. A missing or unwired EN pin
486    // has no decidable guard and is rejected.
487    fn enable<'model>(
488        &mut self,
489        object: &'model FbdObject,
490        survey: &Survey<'model>,
491        broken: &mut Vec<Broken<'model>>,
492        location: &SourceLocation,
493    ) -> Option<AstNode> {
494        if !object.has_execution_control() {
495            return None;
496        }
497
498        let en = object.en_pin();
499        match trace(en.and_then(|pin| pin.source_pin()), survey) {
500            Trace::Reached(source) => {
501                let value = self.value(&source, location);
502                Some(self.negate_if(value, location, en.is_some_and(|pin| pin.negated)))
503            }
504
505            // A consumed chain that never reached a source; reported below.
506            Trace::DeadEnd(at) => {
507                broken.push(at);
508                None
509            }
510
511            Trace::Unwired => {
512                let name = object.type_name().unwrap_or_default();
513                self.diagnostics.push(Diagnostic::unconnected_en(name, location.clone()));
514                None
515            }
516        }
517    }
518
519    fn member_read(&mut self, block: &FbdObject, pin: &Pin, location: &SourceLocation) -> AstNode {
520        let target = block::read_target(block, pin, self.index);
521        let node = st::parse_expression_at(&target, self.ids.clone(), location);
522
523        self.negate_if(node, location, pin.negated)
524    }
525
526    fn negate_if(&mut self, node: AstNode, location: &SourceLocation, negated: bool) -> AstNode {
527        match negated {
528            true => AstFactory::create_not_expression(node, location.clone(), self.ids.next_id()),
529            false => node,
530        }
531    }
532
533    fn parse(&mut self, object: &FbdObject, location: &SourceLocation) -> AstNode {
534        st::parse_expression_at(object.identifier().unwrap_or_default(), self.ids.clone(), location)
535    }

539    fn from(object: &FbdObject) -> Self {
540        match object.kind.as_str() {
541            "ppx:DataSource" => Role::Source,
542            "ppx:DataSink" => Role::Sink,

547            "ppx:Connector" => Role::Connector,
548            "ppx:Continuation" => Role::Continuation,
549            "ppx:Unconnected" => Role::Unconnected,
550            _ => Role::Other,

555impl Broken<'_> {
556    fn diagnostic(&self, factory: &SourceLocationFactory) -> Diagnostic {
557        let location = factory.create_block_location(self.element.global_id);
558        let label = self.element.label().unwrap_or_default();
559
560        match self.reason {
561            Reason::OpenConnector => Diagnostic::open_connector(label, location),
562            Reason::DanglingContinuation => Diagnostic::dangling_continuation(label, location),
563            Reason::EnoCycle => Diagnostic::eno_cycle(self.element.type_name().unwrap_or_default(), location),
564        }
565    }
566}

568fn trace<'model>(start: Option<usize>, survey: &Survey<'model>) -> Trace<'model> {
569    let Some(mut pin) = start else { return Trace::Unwired };
570    let Some(mut element) = survey.by_pin.get(&pin).copied() else { return Trace::Unwired };
571
572    // Trace through the connections until we hit a producer (cycle-guarded).
573    let mut visited = HashSet::new();
574    let mut visited_blocks = HashSet::new();
575    let mut negated = false;
576    loop {
577        if matches!(Role::from(element), Role::Continuation) {
578            let dangling = |element| Trace::DeadEnd(Broken { element, reason: Reason::DanglingContinuation });
579
580            // An unlabeled continuation, or one revisiting a label (a cycle), dangles.
581            let Some(label) = element.label() else { return dangling(element) };
582            if !visited.insert(label) {
583                return dangling(element);
584            }
585
586            // Resolve the label to the connector owning it; an unclaimed label dangles.
587            let Some(connector) = survey.connector_by_label.get(label).copied() else {
588                return dangling(element);
589            };
590
591            // Step onto whatever feeds that connector; an input-less one is open.
592            let open = Broken { element: connector, reason: Reason::OpenConnector };
593            let Some(wire) = consumes(connector) else { return Trace::DeadEnd(open) };
594            let Some(producer) = survey.by_pin.get(&wire).copied() else { return Trace::DeadEnd(open) };
595
596            pin = wire;
597            element = producer;
598        } else if let Some(eno) = element.eno_pin().filter(|eno| eno.output_pin() == Some(pin)) {
599            // An ENO pin mirrors its block's post-negation EN value, negated
600            // once more by its own bubble; step onto the EN wire. A block
601            // revisited on the way is a cycle.
602            if !visited_blocks.insert(element.global_id) {
603                return Trace::DeadEnd(Broken { element, reason: Reason::EnoCycle });
604            }
605
606            // A missing or unwired EN behind the pin is the block's own error
607            // (E152); the consumer drops silently like any unwired wire.
608            let Some(en) = element.en_pin() else { return Trace::Unwired };
609            negated ^= eno.negated ^ en.negated;
610
611            let Some(wire) = en.source_pin() else { return Trace::Unwired };
612            let Some(producer) = survey.by_pin.get(&wire).copied() else { return Trace::Unwired };
613
614            pin = wire;
615            element = producer;
616        } else {
617            break;
618        }
619    }
620
621    // A landed wire is a block output pin (read as a member) or a plain source.
622    match survey.block_output.get(&pin).copied() {
623        Some(output) => Trace::Reached(Source::Output { block: element, pin: output, negated }),
624        None => Trace::Reached(Source::Variable { object: element, negated }),
625    }
626}

628// Sees through parentheses, e.g. `(foo)` or `((5))`.
629fn is_supported(node: &AstNode) -> bool {
630    matches!(node.get_stmt_peeled(), AstStatement::Literal(_) | AstStatement::ReferenceExpr(_))
631}

642
643    pub(super) fn is_stateless(block: &FbdObject, index: &Index) -> bool {
644        block.instance_name.is_none()
645            && block.type_name().and_then(|name| index.find_pou(name)).is_some_and(|pou| pou.is_function())
646    }
647
648    pub(super) fn is_variadic(block: &FbdObject, index: &Index) -> bool {
649        block.type_name().is_some_and(|name| index.get_variadic_member(name).is_some())
650    }
651
652    // The IDE leaves the return pin unnamed; declared outputs always carry their parameter name.
653    pub(super) fn is_return_pin(pin: &Pin) -> bool {
654        pin.parameter_name.is_empty()
655    }
656
657    // The pin's readable name: the callee for the return pin, the parameter otherwise.
658    pub(super) fn pin_name<'pin>(block: &'pin FbdObject, pin: &'pin Pin) -> &'pin str {
659        match is_return_pin(pin) {
660            true => block.type_name().unwrap_or_default(),
661            false => &pin.parameter_name,
662        }
663    }
664
665    pub(super) fn temp_name(block: &FbdObject, pin: &Pin) -> String {
666        format!("__out_{}_{}", pin_name(block, pin), block.global_id)
667    }
668
669    // Every data input; the EN pin guards the call and never passes as an argument.
670    pub(super) fn inputs(block: &FbdObject) -> impl Iterator<Item = &Pin> {
671        let en = block.en_pin();
672        block
673            .input_pins()
674            .iter()
675            .chain(block.inout_pins())
676            .filter(move |pin| !en.is_some_and(|en| std::ptr::eq(en, *pin)))
677    }
678
679    // The captured temporary (stateless) or instance member (stateful).
680    pub(super) fn read_target(block: &FbdObject, pin: &Pin, index: &Index) -> String {
681        match is_stateless(block, index) {
682            true => temp_name(block, pin),
683            false => format!("{}.{}", instance(block).unwrap_or_default(), pin.parameter_name),
684        }
685    }
686
687    // The output's declared type; `None` for a pin the callee doesn't declare.
688    // Outputs pass by reference, so unwrap the auto-deref pointer's inner type.
689    pub(super) fn temp_type(block: &FbdObject, pin: &Pin, index: &Index) -> Option<String> {
690        let function = block.type_name().unwrap_or_default();
691        let variable = match is_return_pin(pin) {
692            true => index.find_return_variable(function),
693            false => index.find_member(function, &pin.parameter_name),
694        }?;
695
696        let information = index.get_type_information_or_void(variable.get_type_name());
697        Some(information.get_inner_name().to_string())
698    }
699
700    // The call operator: `inst`, or `inst.act` for an action.
701    pub(super) fn call_target(block: &FbdObject) -> String {
702        let instance = instance(block).unwrap_or_default();
703        match action(block) {
704            Some(action) => format!("{instance}.{action}"),
705            None => instance.to_string(),
706        }
707    }
708
709    // What outputs are read through: the declared instance, or the owner POU.
710    fn instance(block: &FbdObject) -> Option<&str> {
711        block.instance_name.as_deref().or_else(|| owner(block))
712    }
713
714    // Everything before an action suffix (`MyFb.act` -> `MyFb`).
715    fn owner(block: &FbdObject) -> Option<&str> {
716        block.type_name().map(|name| name.rsplit_once('.').map_or(name, |(owner, _)| owner))
717    }
718
719    // The suffix of a qualified `typeName` (`owner.action`), if any.
720    fn action(block: &FbdObject) -> Option<&str> {
721        block.type_name().and_then(|name| name.rsplit_once('.').map(|(_, action)| action))
722    }
723}

725// Which block output pins some consumer actually reads.
726fn consumed(network: &model::Network, survey: &Survey) -> HashSet<usize> {
727    let mut consumed = HashSet::new();
728    let mut record = |wire: Option<usize>| {
729        if let Trace::Reached(Source::Output { pin, .. }) = trace(wire, survey) {
730            if let Some(id) = pin.output_pin() {
731                consumed.insert(id);
732            }
733        }
734    };
735
736    for object in network.elements() {
737        match Role::from(object) {
738            Role::Sink | Role::Return | Role::Jump => record(consumes(object)),
739            Role::Block => {
740                for pin in block::inputs(object) {
741                    record(pin.source_pin());
742                }
743                if let Some(en) = object.en_pin() {
744                    record(en.source_pin());
745                }
746            }
747            _ => {}
748        }
749    }
750
751    consumed
752}
753
754fn consumes(consumer: &FbdObject) -> Option<usize> {
755    consumer.connection_in.as_ref()?.source_pin()
756}
757
758// Any inversion bubble between a producer and its consumer: the producer's,
759// the consumer's, or one synthesized by an EN hop on the way.
760fn negated(consumer: &FbdObject, source: &Source) -> bool {
761    let negated = match source {
762        Source::Variable { object, negated } => object.out_negated() || *negated,
763        Source::Output { pin, negated, .. } => pin.negated || *negated,
764    };
765
766    negated || consumer.in_negated()
767}

778    fn resolve_project(fixture: &str) -> String {
779        let source = fixture_source(fixture);
780        let pou = Pou::parse(&source.source).unwrap();
781        let ids = IdProvider::default();
782
783        let (interface, _) = crate::st::parse_interface(&pou, &source, ids.clone());
784        let index = crate::test_utils::fixture_index(fixture, &interface);
785
786        let (network, _) = super::Resolver::new(ids, &source, &index).resolve(pou.content().network());
787        render(&network)
788    }

790    // One line per statement, in the network's final evaluation order.
791    fn render(network: &Network) -> String {
792        network
793            .statements
794            .iter()
795            .map(|statement| match statement {
796                Statement::Assignment { sink, source, storage: None } => {
797                    format!("{} := {}", AstSerializer::format(sink), AstSerializer::format(source))
798                }
799                Statement::Assignment { sink, source, storage: Some(storage) } => {
800                    let sink = AstSerializer::format(sink);
801                    let source = AstSerializer::format(source);
802                    match storage {
803                        Storage::Set => format!("IF {source} THEN {sink} := TRUE"),
804                        Storage::Reset => format!("IF {source} THEN {sink} := FALSE"),
805                        Storage::Reference => format!("{sink} REF= {source}"),
806                    }
807                }
808                Statement::Return { condition, .. } => {
809                    format!("RETURN {}", AstSerializer::format(condition))
810                }
811                Statement::Jump { condition: Some(condition), target, .. } => {
812                    format!("JMP {target} IF {}", AstSerializer::format(condition))
813                }
814                Statement::Jump { condition: None, target, .. } => {
815                    format!("JMP {target} <disconnected>")
816                }
817                Statement::Label { name, .. } => format!("LABEL {name}"),
818                Statement::Call { target, arguments, capture, enable, .. } => {
819                    let arguments = arguments
820                        .iter()
821                        .map(|argument| match argument {
822                            Argument::Input { parameter, value } => {
823                                format!("{parameter} := {}", AstSerializer::format(value))
824                            }
825                            Argument::Output { parameter, capture: Some(capture) } => {
826                                format!("{parameter} => {capture}")
827                            }
828                            Argument::Output { parameter, capture: None } => format!("{parameter} => "),
829                            Argument::Variadic { value } => AstSerializer::format(value),
830                        })
831                        .collect::<Vec<_>>()
832                        .join(", ");
833                    let call = match capture {
834                        Some(capture) => format!("{capture} := {target}({arguments})"),
835                        None => format!("{target}({arguments})"),
836                    };
837                    match enable {
838                        Some(enable) => format!("IF {} THEN {call}", AstSerializer::format(enable)),
839                        None => call,
840                    }
841                }
842            })
843            .collect::<Vec<_>>()
844            .join("\n")
845    }
