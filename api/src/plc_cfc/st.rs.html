15// Moves every node of a parsed tree to one location.
16struct Relocate(SourceLocation);
17
18impl AstVisitorMut for Relocate {
19    fn visit(&mut self, node: &mut AstNode) {
20        node.location = self.0.clone();
21
22        // The walker walks the operand of `(foo)` without visiting it, and skips
23        // the name of `.foo` and the elements of `[foo]`.
24        match &mut node.stmt {
25            AstStatement::ParenExpression(operand) => self.visit(operand),
26            AstStatement::ReferenceExpr(ReferenceExpr { access: ReferenceAccess::Global(name), .. }) => {
27                self.visit(name);
28                node.walk(self)
29            }
30            AstStatement::Literal(AstLiteral::Array(Array { elements: Some(elements) })) => {
31                self.visit(elements)
32            }
33            _ => node.walk(self),
34        }
35    }
36}

38pub fn parse_expression(text: &str, ids: IdProvider) -> AstNode {
39    let factory = SourceLocationFactory::internal(text);
40    let mut session = lexer::lex_with_ids(text, ids, factory);
41
42    expressions_parser::parse_expression(&mut session)
43}
44
45// Whether `text` is a single operator keyword that names a builtin function, e.g. `AND`.
46pub fn is_operator_function(text: &str) -> bool {
47    let session = lexer::lex_with_ids(text, IdProvider::default(), SourceLocationFactory::internal(text));
48    session.token.classes().contains(&TokenClass::OperatorFunction) && session.peek() == Token::End
49}

51// A parsed expression relocated to where the network places it, so
52// diagnostics point at the diagram element rather than the parsed text.
53// Every node is relocated, not only the root: the parser locates nested nodes
54// within `text`, which no file contains, so diagnostics on them (e.g. an
55// unresolved `foo`) would print no location at all.
56pub fn parse_expression_at(text: &str, ids: IdProvider, location: &SourceLocation) -> AstNode {
57    let mut node = parse_expression(text, ids);
58    Relocate(location.clone()).visit(&mut node);
59    node
60}

62// The declaration text as the IDE's text editor shows it, completed by the
63// closing keyword the export omits; text locations count lines within it.
64pub fn declaration(pou: &Pou) -> String {
65    let end_keyword = match pou.kind() {
66        PouKind::Function => "END_FUNCTION",
67        PouKind::FunctionBlock => "END_FUNCTION_BLOCK",
68        PouKind::Program => "END_PROGRAM",
69    };
70
71    format!("{}\n{end_keyword}", pou.content().declaration().unwrap_or_default())
72}

74pub fn parse_interface(
75    pou: &Pou,
76    source: &SourceCode,
77    ids: IdProvider,
78) -> (CompilationUnit, Vec<Diagnostic>) {
79    // Parse the completed declaration like any ST source.
80    let declaration = SourceCode { source: declaration(pou), path: source.path.clone() };
81    let factory = SourceLocationFactory::for_source(&declaration);
82    let session = lexer::lex_with_ids(&declaration.source, ids, factory);
83
84    parser::parse(session, LinkageType::Internal, source.get_location_str())
85}
