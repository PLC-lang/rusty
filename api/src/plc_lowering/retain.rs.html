1//! # Retain Variable Lowering
2//!
3//! This module handles the lowering of `RETAIN` variables as defined in IEC 61131-3.
4//! Retain variables persist their values across power cycles by being placed in a
5//! dedicated `.retain` linker section, which a PLC runtime maps to non-volatile memory.
6//!
7//! The lowering pass handles three cases:
8//!
9//! ## 1. PROGRAM retain variables
10//! Variables declared as `VAR RETAIN` inside a `PROGRAM` are extracted into global
11//! retain variables and replaced with auto-dereferenced pointers. For example:
12//!
13//! ```st
14//! PROGRAM Main
15//!     VAR RETAIN
16//!         counter : INT := 0;
17//!     END_VAR
18//! END_PROGRAM
19//! ```
20//!
21//! Is transformed into:
22//! - A global variable `__Main_counter__retain : INT := 0` in a retain block
23//! - The original `counter` becomes an auto-deref pointer to `__Main_counter__retain`
24//!
25//! This indirection ensures the PROGRAM's instance struct contains pointers to the
26//! retain globals, while the actual values live in the `.retain` section.
27//!
28//! ## 2. Global variables with transitive retain
29//! If a global variable's type transitively contains retain members (e.g. an FB with
30//! `VAR RETAIN`), the variable is moved to a retain block even if it was not explicitly
31//! declared in a `VAR_GLOBAL RETAIN` block.
32//!
33//! ## 3. FUNCTION_BLOCK retain variables
34//! Retain variables inside FBs stay in-place within the FB's struct definition. The
35//! entire FB instance receives the `.retain` section treatment when it is instantiated
36//! at the global or program level, handled by the transitive `should_retain()` check
37//! during code generation.
38

53impl RetainParticipant {
54    pub fn new(ids: IdProvider) -> Self {
55        Self { ids }
56    }
57
58    pub fn lower_retain(&mut self, units: &mut [CompilationUnit], index: plc::index::Index) {
59        let mut lowerer = RetainLowerer { ids: self.ids.clone(), index, context: Context::default() };
60        for unit in units {
61            lowerer.visit_compilation_unit(unit);
62        }
63    }
64}
65
66struct RetainLowerer {
67    ids: IdProvider,
68    index: plc::index::Index,
69    context: Context,
70}
71
72#[derive(Debug, Default)]
73struct Context {
74    container_name: Option<String>,
75    container_linkage: Option<LinkageType>,
76    in_program: bool,
77    retain_variables: Vec<Variable>,
78    pointer_types: Vec<UserTypeDeclaration>,
79}
80
81impl AstVisitorMut for RetainLowerer {
82    fn visit_compilation_unit(&mut self, unit: &mut CompilationUnit) {
83        unit.walk(self);
84
85        // Register the pointer types synthesized for extracted program retain variables.
86        unit.user_types.append(&mut self.context.pointer_types);
87
88        // After visiting the compilation unit, add all retain variables to the global vars
89        if !self.context.retain_variables.is_empty() {
90            // Find an existing retain global variable block or create a new one if it doesn't exist
91            unit.global_vars
92                .iter_mut()
93                .find(|block| block.retain)
94                .map(|block| {
95                    block.variables.append(&mut self.context.retain_variables);
96                })
97                .unwrap_or_else(|| {
98                    let retain_block = plc_ast::ast::VariableBlock {
99                        variables: self.context.retain_variables.drain(..).collect(),
100                        kind: plc_ast::ast::VariableBlockType::Global,
101                        constant: false,
102                        retain: true,
103                        linkage: plc_ast::ast::LinkageType::Internal,
104                        location: SourceLocation::internal(),
105                        access: AccessModifier::Public,
106                    };
107                    unit.global_vars.push(retain_block);
108                });
109        }
110    }
111    fn visit_pou(&mut self, pou: &mut plc_ast::ast::Pou) {
112        self.context.in_program = matches!(pou.kind, plc_ast::ast::PouType::Program);
113        self.context.container_name = Some(pou.name.clone());
114        self.context.container_linkage = Some(pou.linkage);
115        pou.walk(self);
116        self.context.in_program = false;
117        self.context.container_name = None;
118        self.context.container_linkage = None;
119    }
120
121    fn visit_variable_block(&mut self, block: &mut plc_ast::ast::VariableBlock) {
122        let variables = std::mem::take(&mut block.variables);
123        //If the block is retain but we are in a program, mark the block as non-retain
124        if block.retain && self.context.in_program {
125            block.retain = false;
126        }
127        for variable in variables {
128            let Some(variable_index) =
129                self.index.find_variable(self.context.container_name.as_deref(), &[variable.get_name()])
130            else {
131                block.variables.push(variable);
132                continue;
133            };
134
135            if !variable_index.should_retain(&self.index) {
136                block.variables.push(variable);
137                continue;
138            }
139
140            if self.context.in_program {
141                let (old_variable, new_var) = self.replace_with_retain_variable(variable);
142                self.context.retain_variables.push(new_var);
143                block.variables.push(old_variable);
144            } else if matches!(block.kind, plc_ast::ast::VariableBlockType::Global) && !block.retain {
145                // Global variable in a non-retain block whose type transitively contains retain
146                // members (e.g. an FB with VAR RETAIN). Move it to a retain block.
147                self.context.retain_variables.push(variable);
148            } else {
149                // FB retain variables stay in-place within the FB's struct. The entire FB instance
150                // gets placed in the .retain section when instantiated at the global/program level,
151                // handled by the transitive should_retain() check during codegen.
152                block.variables.push(variable);
153            }
154        }
155    }
156}

158impl RetainLowerer {
159    /// Replaces a retain variable in a program with a global retain variable and replaces the original variable with an auto reference to the global variable
160    fn replace_with_retain_variable(&mut self, mut variable: Variable) -> (Variable, Variable) {
161        let new_name = format!(
162            "__{}_{}__retain",
163            self.context.container_name.as_deref().unwrap_or_default(),
164            variable.get_name()
165        );
166        // Create a global variable called __<pou_name>_<var_name>__retain and move the initializer and datatype to the global variable
167        let new_var = Variable {
168            name: new_name,
169            data_type_declaration: variable.data_type_declaration.clone(),
170            initializer: variable.initializer.take(),
171            location: variable.location.clone(),
172            address: None,
173        };
174
175        // Register the auto-deref pointer type under its own name. An inline definition would be
176        // extracted by the pre-processor as `__<pou>_<var>`, which clashes with the type of the
177        // same name the pre-processor already created when the variable's type was declared
178        // inline (e.g. an array).
179        let pointer_type_name = format!("{}_ptr", new_var.get_name());
180        let location = variable.data_type_declaration.get_location();
181        let referenced_type = std::mem::replace(
182            &mut variable.data_type_declaration,
183            DataTypeDeclaration::Reference {
184                referenced_type: pointer_type_name.clone(),
185                location: location.clone(),
186            },
187        );
188        self.context.pointer_types.push(UserTypeDeclaration {
189            data_type: DataType::PointerType {
190                name: Some(pointer_type_name),
191                referenced_type: Box::new(referenced_type),
192                auto_deref: Some(AutoDerefType::Alias),
193                type_safe: true,
194                is_function: false,
195            },
196            initializer: None,
197            location,
198            scope: self.context.container_name.clone(),
199            linkage: self.context.container_linkage.unwrap_or(LinkageType::Internal),
200        });
201
202        // Initialize the program variable with a reference to the global retain variable.
203        variable.initializer = Some(AstFactory::create_member_reference(
204            AstFactory::create_identifier(&new_var.name, variable.location.clone(), self.ids.next_id()),
205            None,
206            self.ids.next_id(),
207        ));
208        (variable, new_var)
209    }
210}

218    /// Finds the first global variable block with `retain: true` in a compilation unit.
219    fn find_retain_block(unit: &CompilationUnit) -> &VariableBlock {
220        unit.global_vars.iter().find(|b| b.retain).expect("expected a global retain block")
221    }
222
223    /// Finds a variable by name in a given list of variable blocks.
224    fn find_variable_in_blocks<'a>(blocks: &'a [VariableBlock], name: &str) -> &'a Variable {
225        blocks
226            .iter()
227            .flat_map(|b| &b.variables)
228            .find(|v| v.get_name() == name)
229            .unwrap_or_else(|| panic!("expected variable '{name}' in blocks"))
230    }
231
232    /// Finds a POU by name in a compilation unit and returns a variable from it.
233    fn find_pou_variable<'a>(unit: &'a CompilationUnit, pou_name: &str, var_name: &str) -> &'a Variable {
234        let pou = unit
235            .pous
236            .iter()
237            .find(|p| p.name == pou_name)
238            .unwrap_or_else(|| panic!("POU '{pou_name}' not found"));
239        find_variable_in_blocks(&pou.variable_blocks, var_name)
240    }
241
242    /// Returns the referenced type name from a DataTypeDeclaration, if it is a Reference variant.
243    fn referenced_type_name(decl: &DataTypeDeclaration) -> &str {
244        match decl {
245            DataTypeDeclaration::Reference { referenced_type, .. } => referenced_type,
246            other => panic!("expected DataTypeDeclaration::Reference, got {other:?}"),
247        }
248    }

250    #[test]
251    fn test_retain_lowering_on_program() {
252        let source: SourceCode = r#"
253        PROGRAM Test
254        VAR RETAIN
255            x: INT := 5;
256        END_VAR
257        END_PROGRAM
258        "#
259        .into();
260
261        let (_, project) =
262            parse_and_annotate("test", vec![source]).expect("Failed to parse compilation unit");
263        let unit = project.units[0].get_unit();
264
265        // A global retain block should exist with the extracted retain variable
266        let retain_block = find_retain_block(unit);
267        assert_eq!(retain_block.variables.len(), 1);
268        let retain_var = &retain_block.variables[0];
269        assert_eq!(retain_var.get_name(), "__Test_x__retain");
270        assert_eq!(referenced_type_name(&retain_var.data_type_declaration), "INT");
271        assert!(retain_var.initializer.is_some(), "retain variable should carry the original initializer");
272
273        // The program's variable should be replaced with an auto-deref pointer
274        let pou_var = find_pou_variable(unit, "Test", "x");
275        // The type should now reference the generated pointer type, not INT directly
276        assert_eq!(referenced_type_name(&pou_var.data_type_declaration), "__Test_x__retain_ptr");
277        assert!(pou_var.initializer.is_some(), "program variable should have a reference initializer");
278
279        // The program's variable block should no longer be marked retain
280        let test_pou = unit.pous.iter().find(|p| p.name == "Test").unwrap();
281        assert!(
282            test_pou.variable_blocks.iter().all(|b| !b.retain),
283            "program variable blocks should not be retain after lowering"
284        );
285    }

287    #[test]
288    fn test_retain_lowering_on_program_nested() {
289        let source: SourceCode = r#"
290        FUNCTION_BLOCK FB
291        VAR RETAIN
292            a: INT := 5;
293        END_VAR
294        END_FUNCTION_BLOCK
295        PROGRAM Test
296        VAR
297            x: FB;
298        END_VAR
299        END_PROGRAM
300        "#
301        .into();
302
303        let (_, project) =
304            parse_and_annotate("test", vec![source]).expect("Failed to parse compilation unit");
305        let unit = project.units[0].get_unit();
306
307        // The FB instance should be extracted to a global retain variable
308        let retain_block = find_retain_block(unit);
309        assert_eq!(retain_block.variables.len(), 1);
310        let retain_var = &retain_block.variables[0];
311        assert_eq!(retain_var.get_name(), "__Test_x__retain");
312        assert_eq!(referenced_type_name(&retain_var.data_type_declaration), "FB");
313
314        // The FB's own retain block should remain intact (retain stays in-place for FBs)
315        let fb_pou = unit.pous.iter().find(|p| p.name == "FB").unwrap();
316        let fb_retain_block =
317            fb_pou.variable_blocks.iter().find(|b| b.retain).expect("FB should still have a retain block");
318        let fb_retain_blocks = [fb_retain_block.clone()];
319        let fb_var = find_variable_in_blocks(&fb_retain_blocks, "a");
320        assert_eq!(referenced_type_name(&fb_var.data_type_declaration), "INT");
321
322        // The program's variable should be replaced with an auto-deref pointer to the retain global
323        let pou_var = find_pou_variable(unit, "Test", "x");
324        assert_eq!(referenced_type_name(&pou_var.data_type_declaration), "__Test_x__retain_ptr");
325        assert!(pou_var.initializer.is_some(), "program variable should have a reference initializer");
326    }

329    fn test_retain_lowering_on_program_with_inline_types() {
330        let source: SourceCode = r#"
331        PROGRAM Test
332        VAR RETAIN
333            arr : ARRAY[0..2] OF DINT := [1, 2, 3];
334            sub : DINT(0..100) := 50;
335            str : STRING[20] := 'start';
336            col : (RED, GREEN, BLUE) := GREEN;
337            mat : ARRAY[0..1, 0..2] OF DINT;
338            nst : ARRAY[0..1] OF ARRAY[0..2] OF DINT;
339            rec : STRUCT a : DINT; b : DINT; END_STRUCT
340        END_VAR
341        END_PROGRAM
342        "#
343        .into();
344
345        let (_, project) =
346            parse_and_annotate("test", vec![source]).expect("Failed to parse compilation unit");
347        let unit = project.units[0].get_unit();
348
349        // The extracted inline types and the generated retain pointer types must not clash
350        let mut type_names: Vec<&str> =
351            unit.user_types.iter().filter_map(|it| it.data_type.get_name()).collect();
352        type_names.sort_unstable();
353        let duplicates: Vec<&str> =
354            type_names.windows(2).filter(|pair| pair[0] == pair[1]).map(|pair| pair[0]).collect();
355        assert!(duplicates.is_empty(), "duplicate user types after retain lowering: {duplicates:?}");
356
357        // Every member is extracted into the global retain block
358        let retain_block = find_retain_block(unit);
359        let variable_names: Vec<&str> = retain_block.variables.iter().map(|v| v.get_name()).collect();
360        assert_eq!(
361            variable_names,
362            vec![
363                "__Test_arr__retain",
364                "__Test_sub__retain",
365                "__Test_str__retain",
366                "__Test_col__retain",
367                "__Test_mat__retain",
368                "__Test_nst__retain",
369                "__Test_rec__retain",
370            ]
371        );
372    }

374    #[test]
375    fn test_retain_in_global_nested_should_move_to_retain_block() {
376        let source: SourceCode = r#"
377        FUNCTION_BLOCK FB
378        VAR RETAIN
379            a: INT := 5;
380        END_VAR
381        END_FUNCTION_BLOCK
382        VAR_GLOBAL RETAIN
383            explicit_retain: FB;
384            y : INT;
385        END_VAR
386        VAR_GLOBAL
387            implicit_retain: FB;
388            x : INT;
389        END_VAR
390        "#
391        .into();
392
393        let (_, project) =
394            parse_and_annotate("test", vec![source]).expect("Failed to parse compilation unit");
395        let unit = project.units[0].get_unit();
396
397        // The retain block should contain the explicitly retained variables AND the
398        // implicitly retained one (implicit_retain is FB which transitively contains retain members)
399        let retain_block = find_retain_block(unit);
400        let retain_var_names: Vec<&str> = retain_block.variables.iter().map(|v| v.get_name()).collect();
401        assert!(retain_var_names.contains(&"explicit_retain"), "explicit_retain should be in retain block");
402        assert!(
403            retain_var_names.contains(&"y"),
404            "y should stay in retain block (declared in VAR_GLOBAL RETAIN)"
405        );
406        assert!(
407            retain_var_names.contains(&"implicit_retain"),
408            "implicit_retain should be moved to retain block (transitive retain)"
409        );
410
411        // The non-retain global block should only contain x (a plain INT with no retain)
412        let non_retain_globals: Vec<&str> = unit
413            .global_vars
414            .iter()
415            .filter(|b| !b.retain)
416            .flat_map(|b| &b.variables)
417            .filter(|v| !v.get_name().starts_with("__")) // exclude compiler-generated variables
418            .map(|v| v.get_name())
419            .collect();
420        assert_eq!(non_retain_globals, vec!["x"], "only x should remain in the non-retain global block");
421    }
