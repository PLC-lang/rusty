1//! Lowers generic function calls to concrete monomorphizations.
2//!
3//! For each call to a generic function (e.g. `TO_STRING <T: ANY>` invoked with a `DINT`) this phase
4//! resolves the concrete monomorphization name (`TO_STRING__DINT`), and — when no real POU already
5//! provides it — **materializes** it as an `{external}` POU declaration in the AST, then rewrites
6//! the call to target the concrete name.
7//!
8//! Because monomorphizations become real (external) AST POUs, they survive the subsequent
9//! re-index/re-annotate, downstream passes (aggregate-return lowering, codegen) only ever see
10//! concrete calls, and a genuinely-missing monomorphization becomes a link-time undefined symbol —
11//! uniform for scalar and aggregate returns. This replaces the previous approach where generic
12//! resolution ran inside the annotator's first pass and registered synthetic monomorphs into
13//! `new_index` (which did not survive a re-index).
14//!
15//! Builtin generics (`MUX`/`SEL`/`SHL`/…) are resolved inline by codegen and have no monomorphization,
16//! so their resolution stays in the annotator; this phase handles only non-builtin (user/stdlib)
17//! generic functions. It runs as a `post_annotate` pipeline participant before the aggregate-return
18//! lowerer (see the `PipelineParticipantMut` impl in the driver crate).
19

47/// Resolves a type-name reference in a generic template to its concrete monomorph type.
48///
49/// In the indexed template, a generic parameter `T` on `foo` is referenced by its qualified index
50/// name (`__foo__T`), not the bare symbol — so a direct `subst["T"]` lookup misses. This resolves the
51/// reference through the index: if it is a `Generic { generic_symbol }`, the symbol is what `subst`
52/// (built from the generic map, keyed by symbol) is keyed by. A bare-symbol key is also honoured as a
53/// fallback. Returns `None` when the reference is not a generic to substitute.
54fn resolve_concrete(index: &Index, subst: &FxHashMap<String, String>, type_name: &str) -> Option<String> {
55    if let Some(concrete) = subst.get(type_name) {
56        return Some(concrete.clone());
57    }
58    if let Some(DataTypeInformation::Generic { generic_symbol, .. }) =
59        index.find_effective_type_info(type_name)
60    {
61        return subst.get(generic_symbol).cloned();
62    }
63    None
64}

66/// Substitutes generic-type references in a type declaration with their concrete type names.
67/// `subst` maps a generic symbol (e.g. `"T"`) to a concrete type name (e.g. `"DINT"`); references are
68/// resolved through the index (see [`resolve_concrete`]) and left untouched when not generic. Recurses
69/// through array/pointer/vararg element types so shapes like `ARRAY[..] OF T`, `POINTER TO T` and
70/// `{sized} T...` are substituted in place.
71fn substitute_type_decl(
72    index: &Index,
73    decl: &DataTypeDeclaration,
74    subst: &FxHashMap<String, String>,
75) -> DataTypeDeclaration {
76    match decl {
77        DataTypeDeclaration::Reference { referenced_type, location } => DataTypeDeclaration::Reference {
78            referenced_type: resolve_concrete(index, subst, referenced_type)
79                .unwrap_or_else(|| referenced_type.clone()),
80            location: location.clone(),
81        },
82        DataTypeDeclaration::Aggregate { referenced_type, location } => DataTypeDeclaration::Aggregate {
83            referenced_type: resolve_concrete(index, subst, referenced_type)
84                .unwrap_or_else(|| referenced_type.clone()),
85            location: location.clone(),
86        },
87        DataTypeDeclaration::Definition { data_type, location, scope } => {
88            // A bare generic placeholder collapses to a plain reference to its concrete type.
89            if let DataType::GenericType { generic_symbol, .. } = data_type.as_ref() {
90                if let Some(concrete) = subst.get(generic_symbol) {
91                    return DataTypeDeclaration::Reference {
92                        referenced_type: concrete.clone(),
93                        location: location.clone(),
94                    };
95                }
96            }
97            DataTypeDeclaration::Definition {
98                data_type: Box::new(substitute_data_type(index, data_type, subst)),
99                location: location.clone(),
100                scope: scope.clone(),
101            }
102        }
103    }
104}

106fn substitute_data_type(index: &Index, data_type: &DataType, subst: &FxHashMap<String, String>) -> DataType {
107    match data_type {
108        DataType::ArrayType { name, bounds, referenced_type, is_variable_length } => DataType::ArrayType {
109            name: name.clone(),
110            bounds: bounds.clone(),
111            referenced_type: Box::new(substitute_type_decl(index, referenced_type, subst)),
112            is_variable_length: *is_variable_length,
113        },
114        DataType::PointerType { name, referenced_type, auto_deref, type_safe, is_function } => {
115            DataType::PointerType {
116                name: name.clone(),
117                referenced_type: Box::new(substitute_type_decl(index, referenced_type, subst)),
118                auto_deref: *auto_deref,
119                type_safe: *type_safe,
120                is_function: *is_function,
121            }
122        }
123        DataType::VarArgs { referenced_type, sized } => DataType::VarArgs {
124            referenced_type: referenced_type
125                .as_ref()
126                .map(|it| Box::new(substitute_type_decl(index, it, subst))),
127            sized: *sized,
128        },
129        // Struct/enum/subrange/string/generic-placeholder carry no substitutable element reference
130        // in a generic function signature; clone as-is.
131        other => other.clone(),
132    }
133}

135/// Builds a concrete `{external}` monomorphization (`Pou` + empty-body `Implementation`) from a
136/// generic template by substituting each generic symbol with its concrete type. The result is a
137/// declaration only: codegen emits an external symbol that the linker resolves against the provided
138/// implementation (e.g. the stdlib `.so`) or rejects if genuinely missing. Aggregate return types
139/// are preserved so the aggregate-return lowerer transforms them uniformly afterwards.
140pub(crate) fn materialize_monomorph(
141    index: &Index,
142    template: &Pou,
143    monomorph_name: &str,
144    subst: &FxHashMap<String, String>,
145    id_provider: &mut IdProvider,
146) -> (Pou, Implementation) {
147    // `Pou` is not `Clone`, so rebuild it from the template's (clonable) blocks with substituted
148    // element types, via `new_pou` (which fills the remaining fields with monomorph-appropriate
149    // defaults: no generics, no poly/super/interfaces).
150    let variable_blocks = template
151        .variable_blocks
152        .iter()
153        .map(|block| {
154            let mut block = block.clone();
155            for variable in &mut block.variables {
156                variable.data_type_declaration =
157                    substitute_type_decl(index, &variable.data_type_declaration, subst);
158            }
159            block
160        })
161        .collect();
162
163    let mut pou = new_pou(
164        monomorph_name,
165        id_provider.next_id(),
166        variable_blocks,
167        template.kind.clone(),
168        LinkageType::External,
169        &template.location,
170    );
171    pou.return_type = template.return_type.as_ref().map(|rt| substitute_type_decl(index, rt, subst));
172
173    let implementation = new_implementation(
174        monomorph_name,
175        vec![],
176        pou.kind.clone(),
177        LinkageType::External,
178        template.location.clone(),
179    );
180    (pou, implementation)
181}

183/// Lowering phase that rewrites generic function calls to their concrete monomorphizations and
184/// materializes any missing monomorph as an `{external}` POU declaration.
185///
186/// Runs as a `post_annotate` pipeline participant before the aggregate-return lowerer; the driver
187/// drives the re-index/re-annotate fixed point (see the `PipelineParticipantMut` impl in the driver
188/// crate). By the time this runs, generic resolution no longer happens in the annotator's first pass
189/// — this phase is the single place generics are resolved and monomorphs are created.
190#[derive(Default)]
191pub struct GenericLowerer {
192    pub index: Option<Index>,
193    pub annotation: Option<Box<dyn AnnotationMap>>,
194    pub id_provider: IdProvider,
195    pub diagnostics: Vec<Diagnostic>,
196    /// Monomorph names already resolved this run (materialized or found real) — dedup guard,
197    /// accumulated across fixed-point passes.
198    provided: FxHashSet<String>,
199    /// `(template_name, monomorph_name, symbol->concrete)` collected during a walk, materialized
200    /// afterwards (once the walk's immutable borrow of the units is released).
201    pending: Vec<(String, String, FxHashMap<String, String>)>,
202    /// Operator ids of calls already reported for an `E062` nature violation — dedup guard so a
203    /// left-unresolved invalid call isn't re-reported on subsequent fixed-point passes.
204    reported: FxHashSet<usize>,
205    /// Whether the current pass rewrote a call or queued a monomorph (⇒ re-index and run again).
206    changed: bool,
207}
208
209impl GenericLowerer {
210    pub fn new(id_provider: IdProvider) -> Self {
211        Self { id_provider, ..Default::default() }
212    }
213
214    /// Walks all units rewriting resolvable generic calls to concrete monomorphs and queueing any
215    /// missing monomorph, then materializes the queued monomorphs. Returns `true` if anything
216    /// changed, in which case the caller must re-index/re-annotate and call `lower` again.
217    pub fn lower(&mut self, units: &mut [CompilationUnit]) -> bool {
218        if self.index.is_none() || self.annotation.is_none() {
219            return false;
220        }
221        self.changed = false;
222        self.pending.clear();
223        for unit in units.iter_mut() {
224            self.visit_compilation_unit(unit);
225        }
226        self.materialize_pending(units);
227        self.changed
228    }
229
230    fn materialize_pending(&mut self, units: &mut [CompilationUnit]) {
231        for (template_name, monomorph_name, subst) in std::mem::take(&mut self.pending) {
232            let materialized = {
233                let Some(template) = units.iter().flat_map(|it| &it.pous).find(|it| it.name == template_name)
234                else {
235                    continue;
236                };
237                let index = self.index.as_ref().expect("index set during lowering");
238                materialize_monomorph(index, template, &monomorph_name, &subst, &mut self.id_provider)
239            };
240            if let Some(unit) = units.iter_mut().find(|it| it.pous.iter().any(|p| p.name == template_name)) {
241                unit.pous.push(materialized.0);
242                unit.implementations.push(materialized.1);
243            }
244        }
245    }
246
247    fn lower_call(&mut self, statement: &mut CallStatement) {
248        let resolved = {
249            let index = self.index.as_ref().expect("index set during lowering");
250            let annotations = self.annotation.as_deref().expect("annotations set during lowering");
251
252            let Some(StatementAnnotation::Function { qualified_name, return_type, .. }) =
253                annotations.get(&statement.operator)
254            else {
255                return;
256            };
257            let Some(PouIndexEntry::Function { generics, .. }) = index.find_pou(qualified_name) else {
258                return;
259            };
260            // Non-generic call, or a builtin (MUX/SEL/MOVE/ADD/…) resolved specially by codegen:
261            // leave it untouched.
262            if generics.is_empty() || builtins::get_builtin(qualified_name).is_some() {
263                return;
264            }
265
266            let qualified_name = qualified_name.clone();
267            let return_type = return_type.clone();
268            let generics = generics.clone();
269            let args: Vec<&AstNode> =
270                statement.parameters.as_deref().map(flatten_expression_list).unwrap_or_default();
271            let pairs = paired_args(index, &qualified_name, &args);
272
273            // An argument whose type is still generic (a nested call not yet monomorphized) would
274            // poison the biggest-type derivation; defer this call to a later fixed-point pass.
275            let has_unresolved_generic_arg = pairs.iter().any(|(param_type, arg)| {
276                get_generic_candidate(index, annotations, param_type, arg).is_some_and(|(_, concrete)| {
277                    matches!(
278                        index.find_effective_type_info(concrete),
279                        Some(DataTypeInformation::Generic { .. })
280                    )
281                })
282            });
283            if has_unresolved_generic_arg {
284                return;
285            }
286
287            let candidates = pairs.iter().fold(
288                FxHashMap::<String, Vec<String>>::default(),
289                |mut acc, (param_type, arg)| {
290                    if let Some((symbol, concrete)) =
291                        get_generic_candidate(index, annotations, param_type, arg)
292                    {
293                        acc.entry(symbol.to_string()).or_default().push(concrete.to_string());
294                    }
295                    acc
296                },
297            );
298            let generic_map = derive_generic_types(index, index, &generics, candidates);
299
300            // E062: report each argument whose type does not satisfy its generic parameter's nature.
301            // The hint is the *resolved* monomorph parameter type (from `generic_map`), so that an
302            // e.g. INT argument to an `ANY_REAL` parameter — which resolves to REAL — is allowed,
303            // while a REAL argument to an `ANY_INT` parameter is rejected. This mirrors the check the
304            // validator performed against the first pass's resolved type hint pre-refactor.
305            let mut diagnostics = Vec::new();
306            for (param_type, arg) in &pairs {
307                let Some((symbol, _)) = get_generic_candidate(index, annotations, param_type, arg) else {
308                    continue;
309                };
310                let Some(binding) = generics.iter().find(|it| it.name == symbol) else { continue };
311                let value = match arg.get_stmt() {
312                    AstStatement::Assignment(data) | AstStatement::OutputAssignment(data) => &data.right,
313                    _ => *arg,
314                };
315                let Some(actual) = annotations.get_type(value, index) else { continue };
316                let hint = generic_map
317                    .get(symbol)
318                    .and_then(|ty| index.find_effective_type_by_name(ty.derived_type()))
319                    .or_else(|| annotations.get_type_hint(value, index))
320                    .unwrap_or(actual);
321                if let Some(diagnostic) =
322                    evaluate_generic_nature_violation(actual, hint, binding.nature, index, value)
323                {
324                    diagnostics.push(diagnostic);
325                }
326            }
327
328            // Only lower when every generic parameter resolved to a concrete (non-generic) type;
329            // otherwise this call isn't monomorphizable yet (e.g. an argument is itself generic) and
330            // is left untouched.
331            let fully_resolved = generics.iter().all(|binding| {
332                generic_map.get(&binding.name).is_some_and(|ty| {
333                    !matches!(
334                        index.find_effective_type_info(ty.derived_type()),
335                        Some(DataTypeInformation::Generic { .. })
336                    )
337                })
338            });
339            if !fully_resolved {
340                return;
341            }
342            let (call_name, _) = get_specific_function_annotation(
343                index,
344                &generics,
345                &qualified_name,
346                &return_type,
347                &generic_map,
348                generic_name_resolver,
349            );
350            let already_real = index.find_pou(&call_name).filter(|it| !it.is_generic()).is_some();
351            let subst = generic_map
352                .iter()
353                .map(|(symbol, ty)| (symbol.clone(), ty.derived_type().to_string()))
354                .collect::<FxHashMap<_, _>>();
355            (qualified_name, call_name, subst, already_real, diagnostics)
356        };
357
358        let (template_name, call_name, subst, already_real, diagnostics) = resolved;
359
360        // A nature violation makes the call invalid: report E062 (once, even across fixed-point
361        // passes) and leave the call unresolved. Materializing a nonsensical monomorph and rewriting
362        // the operator would additionally coerce the offending argument and emit spurious downcast
363        // warnings on top of the real error.
364        if !diagnostics.is_empty() {
365            if self.reported.insert(statement.operator.get_id()) {
366                self.diagnostics.extend(diagnostics);
367            }
368            return;
369        }
370
371        // Queue materialization once per monomorph that has no real POU providing it.
372        if self.provided.insert(call_name.clone()) && !already_real {
373            self.pending.push((template_name, call_name.clone(), subst));
374        }
375
376        let location = statement.operator.get_location();
377        *statement.operator =
378            create_member_reference_with_location(&call_name, self.id_provider.clone(), None, location);
379        self.changed = true;
380    }
381}

383/// Pairs each call argument with the declared type name of the parameter it binds to. As in the
384/// annotator and codegen, named arguments claim their slots first, positional arguments fill the
385/// remaining slots left-to-right and surplus positional arguments bind to the variadic member.
386fn paired_args<'a>(index: &Index, pou_name: &str, args: &[&'a AstNode]) -> Vec<(String, &'a AstNode)> {
387    let params = index.get_available_parameters(pou_name);
388
389    let named_positions: FxHashSet<usize> = args
390        .iter()
391        .filter_map(|arg| {
392            let name = arg.get_assignment_identifier()?;
393            params.iter().position(|it| it.get_name().eq_ignore_ascii_case(name))
394        })
395        .collect();
396    let mut free_slots = (0..params.len()).filter(|it| !named_positions.contains(it));
397
398    let mut pairs = Vec::new();
399    for arg in args {
400        if let Some(name) = arg.get_assignment_identifier() {
401            if let Some(param) = params.iter().find(|it| it.get_name().eq_ignore_ascii_case(name)) {
402                pairs.push((param.get_type_name().to_string(), *arg));
403            }
404        } else if let Some(slot) = free_slots.next() {
405            pairs.push((params[slot].get_type_name().to_string(), *arg));
406        } else if let Some(vararg) = index.get_variadic_member(pou_name) {
407            pairs.push((vararg.get_type_name().to_string(), *arg));
408        }
409    }
410    pairs
411}

413impl AstVisitorMut for GenericLowerer {
414    fn visit_compilation_unit(&mut self, unit: &mut CompilationUnit) {
415        if self.index.is_none() {
416            return;
417        }
418        unit.walk(self);
419    }
420
421    fn visit_implementation(&mut self, implementation: &mut plc_ast::ast::Implementation) {
422        // Skip the bodies of generic templates: their calls operate on still-generic types (`T`) and
423        // are never code-generated directly (only their provided monomorphizations are). Lowering
424        // them would mint bogus `FN__T` monomorphs.
425        let is_generic_template = self
426            .index
427            .as_ref()
428            .and_then(|index| index.find_pou(&implementation.name))
429            .is_some_and(|pou| pou.is_generic());
430        if is_generic_template {
431            return;
432        }
433        implementation.walk(self);
434    }
435
436    fn visit_call_statement(&mut self, node: &mut AstNode) {
437        let statement = try_from_mut!(node, CallStatement).expect("CallStatement");
438        // Lower nested/argument calls first (post-order) so an inner monomorph is resolved before the
439        // outer call's argument types are consulted.
440        statement.walk(self);
441        self.lower_call(statement);
442    }
443}

457    fn subst(pairs: &[(&str, &str)]) -> FxHashMap<String, String> {
458        pairs.iter().map(|(k, v)| (k.to_string(), v.to_string())).collect()

505    #[test]
506    fn materializes_external_monomorph_with_substituted_types() {
507        let mut ids = IdProvider::default();
508        // template: FUNCTION foo <T> : T VAR_INPUT x : T; END_VAR
509        let mut template = new_pou(
510            "foo",
511            ids.next_id(),
512            vec![VariableBlock::default()
513                .with_block_type(VariableBlockType::Input(ArgumentProperty::ByVal))
514                .with_variables(vec![new_variable("x", "T")])],
515            PouType::Function,
516            LinkageType::Internal,
517            &SourceLocation::internal(),
518        );
519        template.return_type = Some(reference("T"));
520
521        let (pou, imp) = materialize_monomorph(
522            &Index::default(),
523            &template,
524            "foo__DINT",
525            &subst(&[("T", "DINT")]),
526            &mut ids,
527        );
528
529        assert_eq!(pou.name, "foo__DINT");
530        assert_eq!(pou.linkage, LinkageType::External);
531        assert!(pou.generics.is_empty());
532        assert_eq!(pou.variable_blocks[0].variables[0].data_type_declaration.get_name(), Some("DINT"));
533        assert_eq!(pou.return_type.as_ref().and_then(|it| it.get_name()), Some("DINT"));
534        // declaration only
535        assert_eq!(imp.name, "foo__DINT");
536        assert_eq!(imp.linkage, LinkageType::External);
537        assert!(imp.statements.is_empty());
538    }
