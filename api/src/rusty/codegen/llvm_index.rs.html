11/// Index view containing declared values for the current context
12/// Parent Index is the a fallback lookup index for values not declared locally
13#[derive(Debug, Clone, Default)]
14pub struct LlvmTypedIndex<'ink> {
15    parent_index: Option<&'ink LlvmTypedIndex<'ink>>,
16    type_associations: FxHashMap<String, AnyTypeEnum<'ink>>,
17    pou_type_associations: FxHashMap<String, AnyTypeEnum<'ink>>,
18    global_values: FxHashMap<String, GlobalValue<'ink>>,
19    initial_value_associations: FxHashMap<String, BasicValueEnum<'ink>>,
20    loaded_variable_associations: FxHashMap<String, PointerValue<'ink>>,
21    implementations: FxHashMap<String, FunctionValue<'ink>>,
22    utf08_literals: FxHashMap<String, GlobalValue<'ink>>,
23    utf16_literals: FxHashMap<String, GlobalValue<'ink>>,
24}
25
26pub trait TypeHelper<'ink> {
27    #[allow(clippy::wrong_self_convention)]
28    fn as_basic_type(self) -> Option<BasicTypeEnum<'ink>>;
29}
30
31impl<'ink> TypeHelper<'ink> for AnyTypeEnum<'ink> {
32    fn as_basic_type(self) -> Option<BasicTypeEnum<'ink>> {
33        match self {
34            AnyTypeEnum::ArrayType(value) => Some(value.as_basic_type_enum()),
35            AnyTypeEnum::FloatType(value) => Some(value.as_basic_type_enum()),
36            AnyTypeEnum::IntType(value) => Some(value.as_basic_type_enum()),
37            AnyTypeEnum::PointerType(value) => Some(value.as_basic_type_enum()),
38            AnyTypeEnum::StructType(value) => Some(value.as_basic_type_enum()),
39            AnyTypeEnum::VectorType(value) => Some(value.as_basic_type_enum()),
40            AnyTypeEnum::ScalableVectorType(value) => Some(value.as_basic_type_enum()),
41            AnyTypeEnum::VoidType(_) => None,
42            AnyTypeEnum::FunctionType(_) => None,
43        }
44    }
45}

47impl<'ink> LlvmTypedIndex<'ink> {
48    pub fn create_child(parent: &'ink LlvmTypedIndex<'ink>) -> LlvmTypedIndex<'ink> {
49        LlvmTypedIndex {
50            parent_index: Some(parent),
51            type_associations: FxHashMap::default(),
52            pou_type_associations: FxHashMap::default(),
53            global_values: FxHashMap::default(),
54            initial_value_associations: FxHashMap::default(),
55            loaded_variable_associations: FxHashMap::default(),
56            implementations: FxHashMap::default(),
57            utf08_literals: FxHashMap::default(),
58            utf16_literals: FxHashMap::default(),
59        }
60    }
61
62    pub fn merge(&mut self, mut other: LlvmTypedIndex<'ink>) {
63        for (name, assocication) in other.type_associations.drain() {
64            self.type_associations.insert(name, assocication);
65        }
66        for (name, assocication) in other.pou_type_associations.drain() {
67            self.pou_type_associations.insert(name, assocication);
68        }
69        for (name, assocication) in other.initial_value_associations.drain() {
70            self.initial_value_associations.insert(name, assocication);
71        }
72        for (name, value) in other.global_values.drain() {
73            self.global_values.insert(name, value);
74        }
75        for (name, assocication) in other.initial_value_associations.drain() {
76            self.initial_value_associations.insert(name, assocication);
77        }
78        for (name, assocication) in other.loaded_variable_associations.drain() {
79            self.loaded_variable_associations.insert(name, assocication);
80        }
81        for (name, implementation) in other.implementations.drain() {
82            self.implementations.insert(name, implementation);
83        }
84        self.utf08_literals.extend(other.utf08_literals);
85        self.utf16_literals.extend(other.utf16_literals);
86    }
87
88    pub fn associate_type(
89        &mut self,
90        type_name: &str,
91        target_type: AnyTypeEnum<'ink>,
92    ) -> Result<(), CodegenError> {
93        let name = type_name.to_lowercase();
94
95        log::trace!("registered `{name}` as type `{}`", target_type.print_to_string());
96        self.type_associations.insert(name, target_type);
97        Ok(())
98    }
99
100    pub fn associate_pou_type(
101        &mut self,
102        type_name: &str,
103        target_type: AnyTypeEnum<'ink>,
104    ) -> Result<(), CodegenError> {
105        let name = type_name.to_lowercase();
106
107        log::trace!("registered `{name}` as POU type `{}`", target_type.print_to_string());
108        self.pou_type_associations.insert(name, target_type);
109        Ok(())
110    }
111
112    pub fn associate_initial_value(
113        &mut self,
114        type_name: &str,
115        initial_value: BasicValueEnum<'ink>,
116    ) -> Result<(), CodegenError> {
117        let name = type_name.to_lowercase();
118
119        log::trace!("registered `{name}` as initial value type `{}`", initial_value.print_to_string());
120        self.initial_value_associations.insert(name, initial_value);
121        Ok(())
122    }
123
124    pub fn associate_loaded_local_variable(
125        &mut self,
126        container_name: &str,
127        variable_name: &str,
128        target_value: PointerValue<'ink>,
129    ) -> Result<(), CodegenError> {
130        let name = qualified_name(container_name, variable_name).to_lowercase();
131
132        log::trace!("registered `{name}` as loaded local type `{}`", target_value.print_to_string());
133        self.loaded_variable_associations.insert(name, target_value);
134        Ok(())
135    }
136
137    pub fn associate_global(
138        &mut self,
139        variable_name: &str,
140        global_variable: GlobalValue<'ink>,
141    ) -> Result<(), CodegenError> {
142        let name = variable_name.to_lowercase();
143
144        log::trace!("registered `{name}` as global variable type `{}`", global_variable.print_to_string());
145        self.global_values.insert(name.clone(), global_variable);
146        self.initial_value_associations.insert(name, global_variable.as_pointer_value().into());
147        Ok(())
148    }
149
150    pub fn associate_implementation(
151        &mut self,
152        callable_name: &str,
153        function_value: FunctionValue<'ink>,
154    ) -> Result<(), CodegenError> {
155        let name = callable_name.to_lowercase();
156
157        log::trace!("registered `{name}` as implementation type `{}`", function_value.print_to_string());
158        self.implementations.insert(name, function_value);
159
160        Ok(())
161    }
162
163    pub fn associate_utf08_literal(&mut self, literal: &str, literal_variable: GlobalValue<'ink>) {
164        log::trace!("registered literal {literal}");
165        self.utf08_literals.insert(literal.to_string(), literal_variable);
166    }
167
168    pub fn associate_utf16_literal(&mut self, literal: &str, literal_variable: GlobalValue<'ink>) {
169        log::trace!("registered literal {literal}");
170        self.utf16_literals.insert(literal.to_string(), literal_variable);
171    }
172
173    pub fn find_global_value(&self, name: &str) -> Option<GlobalValue<'ink>> {
174        self.global_values
175            .get(&name.to_lowercase())
176            .copied()
177            .or_else(|| self.parent_index.and_then(|it| it.find_global_value(name)))
178    }
179
180    pub fn find_associated_type(&self, type_name: &str) -> Option<BasicTypeEnum<'ink>> {
181        self.type_associations
182            .get(&type_name.to_lowercase())
183            .copied()
184            .and_then(TypeHelper::as_basic_type)
185            .or_else(|| self.parent_index.and_then(|it| it.find_associated_type(type_name)))
186            .or_else(|| self.find_associated_pou_type(type_name))
187    }
188
189    pub fn find_associated_pou_type(&self, type_name: &str) -> Option<BasicTypeEnum<'ink>> {
190        self.pou_type_associations
191            .get(&type_name.to_lowercase())
192            .copied()
193            .and_then(TypeHelper::as_basic_type)
194            .or_else(|| self.parent_index.and_then(|it| it.find_associated_pou_type(type_name)))
195    }
196
197    pub fn get_associated_type(&self, type_name: &str) -> Result<BasicTypeEnum<'ink>, CodegenError> {
198        self.find_associated_type(type_name)
199            .ok_or_else(|| Diagnostic::unknown_type(type_name, SourceLocation::undefined()).into())
200    }
201
202    pub fn get_associated_pou_type(&self, type_name: &str) -> Result<BasicTypeEnum<'ink>, CodegenError> {
203        self.find_associated_pou_type(type_name)
204            .ok_or_else(|| Diagnostic::unknown_type(type_name, SourceLocation::undefined()).into())
205    }
206
207    pub fn find_associated_initial_value(&self, type_name: &str) -> Option<BasicValueEnum<'ink>> {
208        self.initial_value_associations
209            .get(&type_name.to_lowercase())
210            .copied()
211            .or_else(|| self.parent_index.and_then(|it| it.find_associated_initial_value(type_name)))
212    }
213
214    pub fn find_associated_implementation(&self, callable_name: &str) -> Option<FunctionValue<'ink>> {
215        self.implementations
216            .get(&callable_name.to_lowercase())
217            .copied()
218            .or_else(|| self.parent_index.and_then(|it| it.find_associated_implementation(callable_name)))
219    }
220
221    pub fn find_associated_variable_value(&self, qualified_name: &str) -> Option<BasicValueEnum<'ink>> {
222        self.initial_value_associations
223            .get(&qualified_name.to_lowercase())
224            .copied()
225            .or_else(|| self.parent_index.and_then(|it| it.find_associated_variable_value(qualified_name)))
226    }
227
228    pub fn find_loaded_associated_variable_value(&self, qualified_name: &str) -> Option<PointerValue<'ink>> {
229        let result =
230            self.loaded_variable_associations.get(&qualified_name.to_lowercase()).copied().or_else(|| {
231                self.parent_index.and_then(|it| it.find_loaded_associated_variable_value(qualified_name))
232            });
233
234        //If nothing got associated, see if we have a global we could reuse
235        result.or_else(|| {
236            self.find_associated_variable_value(qualified_name)
237                .filter(|it| it.is_pointer_value())
238                .map(BasicValueEnum::into_pointer_value)
239        })
240    }
241
242    pub fn find_utf08_literal_string(&self, literal: &str) -> Option<&GlobalValue<'ink>> {
243        self.utf08_literals
244            .get(literal)
245            .or_else(|| self.parent_index.and_then(|it| it.find_utf08_literal_string(literal)))
246    }
247
248    pub fn find_utf16_literal_string(&self, literal: &str) -> Option<&GlobalValue<'ink>> {
249        self.utf16_literals
250            .get(literal)
251            .or_else(|| self.parent_index.and_then(|it| it.find_utf16_literal_string(literal)))
252    }
