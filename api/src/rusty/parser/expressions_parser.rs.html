22macro_rules! parse_left_associative_expression {
23    ($lexer: expr, $action : expr,
24        $( $pattern:pat_param )|+,
25    ) => {
26        {
27            let mut left = $action($lexer);
28            while matches!($lexer.token, $( $pattern )|+)  {
29                let operator = match to_operator(&$lexer.token) {
30                    Some(operator) => operator,
31                    None => break,
32                };
33                $lexer.advance();
34                let right = $action($lexer);
35                left = AstFactory::create_binary_expression(left, operator, right, $lexer.next_id());
36            }
37            left
38        }
39    };
40}

42/// parse_expression(): returns expression as Statement. if a parse error
43/// is encountered, the erroneous part of the AST will consist of an
44/// EmptyStatement and a diagnostic will be logged. That case is different from
45/// only an EmptyStatement returned, which does not denote an error condition.
46pub fn parse_expression(lexer: &mut ParseSession) -> AstNode {
47    if lexer.token == KeywordSemicolon {
48        AstFactory::create_empty_statement(lexer.location(), lexer.next_id())
49    } else {
50        parse_expression_list(lexer)
51    }
52}
53
54pub fn parse_expression_list(lexer: &mut ParseSession) -> AstNode {
55    let start = lexer.location();
56    let left = parse_range_statement(lexer);
57    if lexer.token == KeywordComma {
58        let mut expressions = vec![];
59        // this starts an expression list
60        while lexer.token == KeywordComma {
61            lexer.advance();
62            if !lexer.closes_open_region(&lexer.token) {
63                expressions.push(parse_range_statement(lexer));
64            }
65        }
66        // we may have parsed no additional expression because of trailing comma
67        if !expressions.is_empty() {
68            expressions.insert(0, left);
69            return AstFactory::create_expression_list(
70                expressions,
71                start.span(&lexer.last_location()),
72                lexer.next_id(),
73            );
74        }
75    }
76    left
77}

79pub(crate) fn parse_range_statement(lexer: &mut ParseSession) -> AstNode {
80    let start = parse_or_expression(lexer);
81
82    if lexer.token == KeywordDotDot {
83        lexer.advance();
84        let end = parse_or_expression(lexer);
85        return AstFactory::create_range_statement(start, end, lexer.next_id());
86    }
87    start
88}

137// Expoent **
138fn parse_exponent_expression(lexer: &mut ParseSession) -> AstNode {
139    //This is always parsed as a function call to the EXPT function
140    //Parse left
141    let mut left = parse_unary_expression(lexer);
142    while matches!(lexer.token, OperatorExponent) {
143        lexer.advance();
144        let right = parse_unary_expression(lexer);
145        let span = left.get_location().span(&right.get_location());
146        left =
147            AstFactory::create_call_to_with_ids("EXPT", vec![left, right], &span, lexer.id_provider.clone());
148    }
149    left
150}

152// UNARY -x, NOT x
153fn parse_unary_expression(lexer: &mut ParseSession) -> AstNode {
154    // collect all consecutive operators
155    let start_location = lexer.location();
156    let mut operators = vec![];
157    while let Some(operator) = match lexer.token {
158        OperatorNot => Some(Operator::Not),
159        OperatorPlus => Some(Operator::Plus),
160        OperatorMinus => Some(Operator::Minus),
161        _ => None,
162    } {
163        operators.push(operator);
164        lexer.advance();
165    }
166    // created nested statements if necessary (e.g. &&)
167    let init = parse_leaf_expression(lexer);
168    operators.iter().rev().fold(init, |expression, operator| {
169        let expression_location = expression.get_location();
170        let location = start_location.span(&expression_location);
171
172        match (&operator, &expression.get_stmt()) {
173            (Operator::Minus, AstStatement::Literal(AstLiteral::Integer(value))) => {
174                AstNode::new_literal(AstLiteral::new_integer(-value), lexer.next_id(), location)
175            }
176
177            (Operator::Plus, AstStatement::Literal(AstLiteral::Integer(value))) => {
178                AstNode::new_literal(AstLiteral::new_integer(*value), lexer.next_id(), location)
179            }
180
181            // Handle negative and positive real literals directly instead of creating UnaryExpressions
182            // This ensures that the minus sign is part of the literal string, which is crucial for
183            // proper constant evaluation (especially for very small numbers like 1e-100)
184            (Operator::Minus, AstStatement::Literal(AstLiteral::Real(value))) => {
185                let negative_value = format!("-{}", value);
186                AstNode::new_literal(AstLiteral::new_real(negative_value), lexer.next_id(), location)
187            }
188
189            (Operator::Plus, AstStatement::Literal(AstLiteral::Real(value))) => {
190                AstNode::new_literal(AstLiteral::new_real(value.clone()), lexer.next_id(), location)
191            }
192
193            // Return the reference itself instead of wrapping it inside a `AstStatement::UnaryExpression`
194            (Operator::Plus, AstStatement::Identifier(name)) => {
195                AstFactory::create_identifier(name, &location, lexer.next_id())
196            }
197
198            _ => AstFactory::create_unary_expression(*operator, expression, location, lexer.next_id()),
199        }
200    })
201}

203fn to_operator(token: &Token) -> Option<Operator> {
204    match token {
205        OperatorPlus => Some(Operator::Plus),
206        OperatorMinus => Some(Operator::Minus),
207        OperatorMultiplication => Some(Operator::Multiplication),
208        OperatorExponent => Some(Operator::Exponentiation),
209        OperatorDivision => Some(Operator::Division),
210        OperatorEqual => Some(Operator::Equal),
211        OperatorNotEqual => Some(Operator::NotEqual),
212        OperatorLess => Some(Operator::Less),
213        OperatorGreater => Some(Operator::Greater),
214        OperatorLessOrEqual => Some(Operator::LessOrEqual),
215        OperatorGreaterOrEqual => Some(Operator::GreaterOrEqual),
216        OperatorModulo => Some(Operator::Modulo),
217        OperatorAnd | OperatorAmp => Some(Operator::And),
218        OperatorAndThen => Some(Operator::AndThen),
219        OperatorOr => Some(Operator::Or),
220        OperatorOrElse => Some(Operator::OrElse),
221        OperatorXor => Some(Operator::Xor),
222        OperatorNot => Some(Operator::Not),
223        _ => None,
224    }
225}

227// Literals, Identifiers, etc.
228fn parse_leaf_expression(lexer: &mut ParseSession) -> AstNode {
229    let literal_parse_result = match lexer.token {
230        OperatorMultiplication => parse_vla_range(lexer),
231        _ => parse_call_statement(lexer),
232    };
233
234    match literal_parse_result {
235        Some(statement) => match lexer.token {
236            KeywordAssignment => {
237                lexer.advance();
238                AstFactory::create_assignment(statement, parse_range_statement(lexer), lexer.next_id())
239            }
240            KeywordOutputAssignment => {
241                lexer.advance();
242                AstFactory::create_output_assignment(statement, parse_range_statement(lexer), lexer.next_id())
243            }
244            KeywordReferenceAssignment => {
245                lexer.advance();
246                AstFactory::create_ref_assignment(statement, parse_range_statement(lexer), lexer.next_id())
247            }
248            _ => statement,
249        },
250        None => {
251            let statement = AstFactory::create_empty_statement(
252                lexer.diagnostics.last().map_or(SourceLocation::undefined(), |d| d.get_location()),
253                lexer.next_id(),
254            );
255            statement
256        }
257    }
258}

260/// parse an expression at the bottom of the parse-tree.
261/// leaf-expressions are literals, identifier, direct-access and parenthesized expressions
262/// (since the parentheses change the parse-priority)
263fn parse_atomic_leaf_expression(lexer: &mut ParseSession<'_>) -> Option<AstNode> {
264    // Check if we're dealing with a number that has an explicit '+' or '-' sign...
265
266    match lexer.token {
267        OperatorPlus | OperatorMinus => {
268            let is_negative = lexer.token == OperatorMinus;
269            lexer.advance();
270
271            match lexer.token {
272                LiteralInteger => parse_literal_number(lexer, is_negative),
273                LiteralIntegerBin => parse_literal_number_with_modifier(lexer, 2, is_negative),
274                LiteralIntegerOct => parse_literal_number_with_modifier(lexer, 8, is_negative),
275                LiteralIntegerHex => parse_literal_number_with_modifier(lexer, 16, is_negative),
276                _ => {
277                    lexer.accept_diagnostic(Diagnostic::unexpected_token_found(
278                        "Numeric Literal",
279                        lexer.slice(),
280                        lexer.location(),
281                    ));
282                    None
283                }
284            }
285        }
286        KeywordParensOpen => {
287            parse_any_in_region(lexer, vec![KeywordParensClose], |lexer| {
288                lexer.advance(); // eat KeywordParensOpen
289
290                let start = lexer.last_location();
291                let expr = parse_expression(lexer);
292
293                Some(AstFactory::create_paren_expression(
294                    expr,
295                    start.span(&lexer.location()),
296                    lexer.next_id(),
297                ))
298            })
299        }
300        token if token.is_identifier_like() => Some(parse_identifier(lexer)),
301        KeywordSuper => {
302            lexer.advance();
303            Some(AstFactory::create_super_reference(
304                lexer.last_location(),
305                lexer.try_consume(OperatorDeref).then_some(()),
306                lexer.next_id(),
307            ))
308        }
309        KeywordThis => {
310            lexer.advance();
311            Some(AstFactory::create_this_reference(lexer.last_location(), lexer.next_id()))
312        }
313        HardwareAccess((hw_type, access_type)) => parse_hardware_access(lexer, hw_type, access_type),
314        LiteralInteger => parse_literal_number(lexer, false),
315        LiteralIntegerBin => parse_literal_number_with_modifier(lexer, 2, false),
316        LiteralIntegerOct => parse_literal_number_with_modifier(lexer, 8, false),
317        LiteralIntegerHex => parse_literal_number_with_modifier(lexer, 16, false),
318        LiteralDate => parse_literal_date(lexer),
319        LiteralTimeOfDay => parse_literal_time_of_day(lexer),
320        LiteralTime => parse_literal_time(lexer),
321        LiteralDateAndTime => parse_literal_date_and_time(lexer),
322        LiteralString => parse_literal_string(lexer, false),
323        LiteralWideString => parse_literal_string(lexer, true),
324        LiteralTrue => parse_bool_literal(lexer, true),
325        LiteralFalse => parse_bool_literal(lexer, false),
326        LiteralNull => parse_null_literal(lexer),
327        KeywordSquareParensOpen => parse_array_literal(lexer),
328        DirectAccess(access) => parse_direct_access(lexer, access),
329        _ => {
330            if lexer.closing_keywords.contains(&vec![KeywordParensClose])
331                && matches!(lexer.last_token, KeywordOutputAssignment | KeywordAssignment)
332                && matches!(lexer.token, KeywordParensClose | KeywordComma)
333            {
334                // due to closing keyword ')' and last_token '=>' / ':='
335                // we are probably in a call statement missing a parameter assignment 'foo(param := );
336                // optional parameter assignments are allowed, validation should handle any unwanted cases
337                Some(AstFactory::create_empty_statement(lexer.location(), lexer.next_id()))
338            } else {
339                lexer.accept_diagnostic(Diagnostic::unexpected_token_found(
340                    "expression",
341                    lexer.slice(),
342                    lexer.location(),
343                ));
344                // If the bad token has any operator interpretation, drain
345                // any consecutive operator tokens and retry the leaf so the
346                // recovered AST captures the operand the user wrote. In
347                // practice this fires for *binary-only* operators misused
348                // as a prefix (e.g. `&y`, `MOD y`); unary-eligible
349                // operators (`-`, `+`, `NOT`) also match `to_operator()`
350                // but normally never reach this fallback because the
351                // prefix-operator parser higher in the cascade handles
352                // them first. Looping (rather than tail-recursing per
353                // token) emits one diagnostic per run instead of N, and
354                // keeps the recovery cost flat on adversarial input like
355                // `& & & g`. For non-operator tokens (e.g. `END_CASE`,
356                // `END_VAR`), leave the token in the stream so outer
357                // parsers can synchronise on it — the `advanced` guard
358                // prevents an infinite retry loop in that case.
359                let mut advanced = false;
360                while to_operator(&lexer.token).is_some() {
361                    lexer.advance();
362                    advanced = true;
363                }
364                if advanced {
365                    return parse_atomic_leaf_expression(lexer);
366                }
367                None
368            }
369        }
370    }
371}

382fn parse_array_literal(lexer: &mut ParseSession) -> Option<AstNode> {
383    let start = lexer.range().start;
384    expect_token!(lexer, KeywordSquareParensOpen, None);
385    lexer.advance();
386    let elements = Some(Box::new(parse_expression(lexer)));
387    let end = lexer.range().end;
388    expect_token!(lexer, KeywordSquareParensClose, None);
389    lexer.advance();
390
391    Some(AstNode::new_literal(
392        AstLiteral::new_array(elements),
393        lexer.next_id(),
394        lexer.source_range_factory.create_range(start..end),
395    ))
396}

415pub fn parse_call_statement(lexer: &mut ParseSession) -> Option<AstNode> {
416    let reference = parse_qualified_reference(lexer)?;
417    let reference_loc = reference.get_location();
418
419    // We're not dealing with a call statement here
420    if !lexer.try_consume(KeywordParensOpen) {
421        return Some(reference);
422    }
423
424    let call = if lexer.try_consume(KeywordParensClose) {
425        AstFactory::create_call_statement(
426            reference,
427            None,
428            lexer.next_id(),
429            reference_loc.span(&lexer.location()),
430        )
431    } else {
432        parse_any_in_region(lexer, vec![KeywordParensClose], |lexer| {
433            AstFactory::create_call_statement(
434                reference,
435                Some(parse_expression_list(lexer)),
436                lexer.next_id(),
437                reference_loc.span(&lexer.location()),
438            )
439        })
440    };
441
442    // Postfix chain after the call: `foo()[...]`, `foo()^`, or combinations
443    // like `foo()[...]^` and `foo()^[...]`.
444    let mut result = call;
445    loop {
446        if lexer.try_consume(KeywordSquareParensOpen) {
447            let index = parse_any_in_region(lexer, vec![KeywordSquareParensClose], parse_expression);
448            result = AstFactory::create_index_reference(
449                index,
450                Some(result),
451                lexer.next_id(),
452                SourceLocation::undefined(),
453            );
454        } else if lexer.try_consume(OperatorDeref) {
455            let deref_loc = lexer.last_location();
456            result =
457                AstFactory::create_deref_reference(result, lexer.next_id(), reference_loc.span(&deref_loc));
458        } else {
459            break;
460        }
461    }
462    Some(result)
463}

465pub fn parse_qualified_reference(lexer: &mut ParseSession) -> Option<AstNode> {
466    let mut current = None;
467    let mut pos = lexer.parse_progress - 1; // force an initial loop
468
469    // as long as we parse something we keep eating stuff eagerly
470    while lexer.parse_progress > pos {
471        pos = lexer.parse_progress;
472        match (
473            current,
474            // only test for the tokens without eating it
475            [KeywordDot, KeywordSquareParensOpen, OperatorDeref, TypeCastPrefix]
476                .into_iter()
477                .find(|it| lexer.token == *it),
478        ) {
479            // No base, No token -> Beginning of a qualified reference
480            (None, None) => {
481                let exp = parse_atomic_leaf_expression(lexer)?;
482                // pack if this is something to be resolved
483                current = if exp.is_identifier() {
484                    Some(AstFactory::create_member_reference(exp, None, lexer.next_id()))
485                } else {
486                    Some(exp)
487                };
488            }
489            // Global Namespace Operator, e.g. `.foo`
490            (None, Some(KeywordDot)) => {
491                let location_dot = lexer.location();
492                lexer.advance();
493
494                let expr = parse_atomic_leaf_expression(lexer)?;
495                let location = location_dot.span(&expr.location);
496
497                current = Some(AstFactory::create_global_reference(lexer.next_id(), expr, location));
498            }
499            // base._ -> a segment of a qualified reference, we stand right after the dot
500            (Some(base), Some(KeywordDot)) => {
501                lexer.advance();
502                let member = if lexer.token == LiteralInteger {
503                    let index = parse_strict_literal_integer(lexer)?;
504                    let location = index.get_location();
505                    AstFactory::create_direct_access(DirectAccessType::Bit, index, lexer.next_id(), location)
506                } else {
507                    parse_atomic_leaf_expression(lexer)?
508                };
509                current = Some(AstFactory::create_member_reference(member, Some(base), lexer.next_id()));
510            }
511            // CAST-Statement: INT#a.b.c
512            // this means INT#(a.b.c) rather than (INT#a).b.c
513            (_, Some(TypeCastPrefix)) => {
514                let location_start = lexer.range().start;
515                let location = lexer.location();
516                let mut type_name = lexer.slice_and_advance();
517                type_name.pop(); // get rid of the "#" at the end
518                let stmt = parse_atomic_leaf_expression(lexer)?;
519                let end = stmt.get_location();
520                let type_range = lexer
521                    .source_range_factory
522                    .create_range(location_start..(location_start + type_name.len()));
523                current = Some(AstFactory::create_cast_statement(
524                    AstFactory::create_member_reference(
525                        AstFactory::create_identifier(type_name.as_str(), &type_range, lexer.next_id()),
526                        None,
527                        lexer.next_id(),
528                    ),
529                    stmt,
530                    &location.span(&end),
531                    lexer.next_id(),
532                ));
533            }
534            (Some(base), Some(KeywordSquareParensOpen)) => {
535                lexer.advance();
536                let index_reference =
537                    parse_any_in_region(lexer, vec![KeywordSquareParensClose], parse_expression);
538                let new_location = base.get_location().span(&lexer.last_location());
539                current = Some({
540                    AstFactory::create_index_reference(
541                        index_reference,
542                        Some(base),
543                        lexer.next_id(),
544                        new_location,
545                    )
546                })
547            }
548            (Some(base), Some(OperatorDeref)) => {
549                lexer.advance();
550                let new_location = base.get_location().span(&lexer.last_location());
551                current = Some(AstFactory::create_deref_reference(base, lexer.next_id(), new_location))
552            }
553            (last_current, _) => {
554                current = last_current; // exit the loop
555            }
556        }
557    }
558    match current {
559        Some(current) => Some(current),
560        None => parse_atomic_leaf_expression(lexer),
561    }
562}

564fn parse_direct_access(lexer: &mut ParseSession, access: DirectAccessType) -> Option<AstNode> {
565    //Consume the direct access
566    let location = lexer.location();
567    lexer.advance();
568    //The next token can either be an integer or an identifier
569    let index = match lexer.token {
570        LiteralInteger => parse_strict_literal_integer(lexer),
571        token if token.is_identifier_like() => {
572            let location = lexer.location();
573            Some(AstFactory::create_member_reference(
574                AstFactory::create_identifier(lexer.slice_and_advance().as_str(), location, lexer.next_id()),
575                None,
576                lexer.next_id(),
577            ))
578        }
579        _ => {
580            lexer.accept_diagnostic(Diagnostic::unexpected_token_found(
581                "Integer or Reference",
582                lexer.slice(),
583                lexer.location(),
584            ));
585            None
586        }
587    }?;
588    let location = location.span(&lexer.last_location());
589    Some(AstFactory::create_direct_access(access, index, lexer.next_id(), location))
590}

592fn parse_literal_number_with_modifier(
593    lexer: &mut ParseSession,
594    radix: u32,
595    is_negative: bool,
596) -> Option<AstNode> {
597    // we can safely unwrap the number string, since the token has
598    // been matched using regular expressions
599    let location = lexer.location();
600    let token = lexer.slice_and_advance();
601    let number_str = token.split('#').next_back().expect("token with '#'");
602    let number_str = number_str.replace('_', "");
603
604    // again, the parsed number can be safely unwrapped.
605    let value = i128::from_str_radix(number_str.as_str(), radix).expect("valid i128");
606    let value = if is_negative { -value } else { value };
607    Some(AstNode::new_literal(AstLiteral::new_integer(value), lexer.next_id(), location))
608}

610fn parse_literal_number(lexer: &mut ParseSession, is_negative: bool) -> Option<AstNode> {
611    let location = if is_negative {
612        //correct the location if we just parsed a minus before
613        lexer.last_range.start..lexer.range().end
614    } else {
615        lexer.range()
616    };
617    let result = lexer.slice_and_advance();
618    if result.to_lowercase().contains('e') {
619        let value = result.replace('_', "");
620        //Treat exponents as reals
621        return Some(AstNode::new_literal(
622            AstLiteral::new_real(value),
623            lexer.next_id(),
624            lexer.source_range_factory.create_range(location),
625        ));
626    } else if lexer.try_consume(KeywordDot) {
627        return parse_literal_real(lexer, result, location, is_negative);
628    } else if lexer.try_consume(KeywordParensOpen) {
629        let start = location.start;
630        let multiplier = match result.parse::<u32>() {
631            Ok(v) => Some(v),
632            Err(e) => {
633                lexer.accept_diagnostic(
634                    Diagnostic::new(format!("Failed to parse number {result}"))
635                        .with_error_code("E011")
636                        .with_location(lexer.source_range_factory.create_range(location))
637                        .with_internal_error(e.into()),
638                );
639                None
640            }
641        }?;
642        let element = parse_expression(lexer);
643        expect_token!(lexer, KeywordParensClose, None);
644        let end = lexer.range().end;
645        lexer.advance();
646        return Some(AstFactory::create_multiplied_statement(
647            multiplier,
648            element,
649            lexer.source_range_factory.create_range(start..end),
650            lexer.next_id(),
651        ));
652    }
653
654    // parsed number value can be safely unwrapped
655    let result = result.replace('_', "");
656
657    let value = result.parse::<i128>().expect("valid i128");
658    let value = if is_negative { -value } else { value };
659
660    Some(AstNode::new_literal(
661        AstLiteral::new_integer(value),
662        lexer.next_id(),
663        lexer.source_range_factory.create_range(location),
664    ))
665}

667/// Parses a literal integer without considering Signs or the Possibility of a Floating Point/ Exponent
668pub fn parse_strict_literal_integer(lexer: &mut ParseSession) -> Option<AstNode> {
669    //correct the location if we just parsed a minus before
670    let location = lexer.location();
671    let result = lexer.slice_and_advance();
672    // parsed number value can be safely unwrapped
673    let result = result.replace('_', "");
674    if result.to_lowercase().contains('e') {
675        lexer.accept_diagnostic(Diagnostic::unexpected_token_found(
676            "Integer",
677            &format!("Exponent value: {result}"),
678            location,
679        ));
680        None
681    } else {
682        let value = result.parse::<i128>().expect("valid i128");
683        Some(AstNode::new_literal(AstLiteral::new_integer(value), lexer.next_id(), location))
684    }
685}

691/// Errors produced by [`handle_special_chars`] for invalid `$`-escape sequences.
692#[derive(Debug, PartialEq, Clone)]
693enum EscapeError {
694    /// `$` at the very end of the string — nothing follows it.
695    ///
696    /// **Note**: this variant is currently unreachable from [`parse_literal_string`].
697    /// The lexer regex treats `$.` (dollar + any character) as an atomic unit, so a
698    /// `$` immediately before the closing delimiter is consumed as the `$'`/`$"` quote-escape,
699    /// leaving the string unterminated — a lexer-level E007 rather than E124.
700    /// Detecting this case as E124 requires the lexer to distinguish mid-string `$'`
701    /// (quote-escape) from trailing `$` + closing delimiter, which needs a separate
702    /// token variant and is tracked as a future enhancement.
703    TrailingDollar,
704    /// `$X` where `X` is not a recognised named-escape character and not a hex digit.
705    UnrecognizedEscape(char),
706    /// `$` followed by hex digit(s) but not enough to form a complete escape
707    /// (STRING needs 2, WSTRING needs 4).
708    IncompleteHexEscape,
709}
710
711/// Process all `$`-escape sequences in a string literal (content between quotes) in a
712/// single pass.  Returns the decoded string and a list of any invalid escapes found.
713/// Invalid escapes are kept as-is in the decoded string for error-recovery purposes.
714fn handle_special_chars(string: &str, is_wide: bool) -> (String, Vec<EscapeError>) {
715    let hex_digits_per_unit = if is_wide { 4 } else { 2 };
716
717    let mut result = String::with_capacity(string.len());
718    let mut errors: Vec<EscapeError> = Vec::new();
719    let chars: Vec<char> = string.chars().collect();
720    let len = chars.len();
721    let mut i = 0;
722
723    while i < len {
724        if chars[i] != '$' {
725            result.push(chars[i]);
726            i += 1;
727            continue;
728        }
729
730        // We have a '$' — check what follows.
731        if i + 1 >= len {
732            errors.push(EscapeError::TrailingDollar);
733            result.push('$');
734            i += 1;
735            continue;
736        }
737
738        let next = chars[i + 1];
739
740        // Named escapes
741        match next {
742            'l' | 'L' | 'n' | 'N' => {
743                result.push('\n');
744                i += 2;
745                continue;
746            }
747            'p' | 'P' => {
748                result.push('\x0C');
749                i += 2;
750                continue;
751            }
752            'r' | 'R' => {
753                result.push('\r');
754                i += 2;
755                continue;
756            }
757            't' | 'T' => {
758                result.push('\t');
759                i += 2;
760                continue;
761            }
762            '$' => {
763                result.push('$');
764                i += 2;
765                continue;
766            }
767            '\'' | '"' => {
768                result.push(next);
769                i += 2;
770                continue;
771            }
772            _ => {}
773        }
774
775        // Try to consume consecutive hex escape sequences ($XX or $XXXX).
776        if next.is_ascii_hexdigit() {
777            if is_wide {
778                let mut hex_units: Vec<u16> = Vec::new();
779                while i < len && chars[i] == '$' {
780                    if i + 1 + hex_digits_per_unit > len {
781                        break;
782                    }
783                    let hex_str: String = chars[i + 1..i + 1 + hex_digits_per_unit].iter().collect();
784                    if hex_str.len() == hex_digits_per_unit && hex_str.chars().all(|c| c.is_ascii_hexdigit())
785                    {
786                        hex_units.push(u16::from_str_radix(&hex_str, 16).unwrap_or_default());
787                        i += 1 + hex_digits_per_unit;
788                    } else {
789                        break;
790                    }
791                }
792                if !hex_units.is_empty() {
793                    result.push_str(&String::from_utf16_lossy(&hex_units));
794                    continue;
795                }
796            } else {
797                let mut hex_bytes: Vec<u8> = Vec::new();
798                while i < len && chars[i] == '$' {
799                    if i + 1 + hex_digits_per_unit > len {
800                        break;
801                    }
802                    let hex_str: String = chars[i + 1..i + 1 + hex_digits_per_unit].iter().collect();
803                    if hex_str.len() == hex_digits_per_unit && hex_str.chars().all(|c| c.is_ascii_hexdigit())
804                    {
805                        hex_bytes.push(u8::from_str_radix(&hex_str, 16).unwrap_or_default());
806                        i += 1 + hex_digits_per_unit;
807                    } else {
808                        break;
809                    }
810                }
811                if !hex_bytes.is_empty() {
812                    result.push_str(&String::from_utf8_lossy(&hex_bytes));
813                    continue;
814                }
815            }
816            // next was a hex digit but we couldn't form a complete escape.
817            errors.push(EscapeError::IncompleteHexEscape);
818            result.push('$');
819            i += 1;
820            continue;
821        }
822
823        // Unrecognized escape — not a named escape and not a hex digit.
824        errors.push(EscapeError::UnrecognizedEscape(next));
825        result.push('$');
826        i += 1;
827    }
828
829    (result, errors)
830}

832fn parse_literal_string(lexer: &mut ParseSession, is_wide: bool) -> Option<AstNode> {
833    let result = lexer.slice();
834    let location = lexer.location();
835
836    let (processed, errors) = handle_special_chars(&trim_quotes(result), is_wide);
837
838    for error in &errors {
839        let message = match error {
840            EscapeError::TrailingDollar => {
841                "Invalid escape sequence in string literal: trailing '$' has nothing to escape".to_string()
842            }
843            EscapeError::UnrecognizedEscape(c) => {
844                format!("Invalid escape sequence in string literal: '${c}' is not a valid escape sequence")
845            }
846            EscapeError::IncompleteHexEscape => {
847                let n = if is_wide { 4 } else { 2 };
848                format!(
849                    "Invalid escape sequence in string literal: incomplete hex escape, expected {n} hex digits after '$'"
850                )
851            }
852        };
853        lexer.accept_diagnostic(
854            Diagnostic::new(message).with_error_code("E124").with_location(location.clone()),
855        );
856    }
857
858    let string_literal =
859        Some(AstNode::new_literal(AstLiteral::new_string(processed, is_wide), lexer.next_id(), location));
860    lexer.advance();
861    string_literal
862}
863
864fn parse_literal_real(
865    lexer: &mut ParseSession,
866    integer: String,
867    integer_range: Range<usize>,
868    is_negative: bool,
869) -> Option<AstNode> {
870    if lexer.token == LiteralInteger {
871        let start = integer_range.start;
872        let end = lexer.range().end;
873        let fractional = lexer.slice_and_advance();
874        let integer = integer.replace('_', "");
875        let value = format!("{}{}.{}", if is_negative { "-" } else { "" }, integer, fractional);
876        let new_location = lexer.source_range_factory.create_range(start..end);
877
878        Some(AstNode::new_literal(AstLiteral::new_real(value), lexer.next_id(), new_location))
879    } else {
880        lexer.accept_diagnostic(Diagnostic::unexpected_token_found(
881            "LiteralInteger or LiteralExponent",
882            lexer.slice(),
883            lexer.location(),
884        ));
885        None
886    }
887}
