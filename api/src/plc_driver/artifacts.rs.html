1//! Naming of the intermediate build artifacts (object, bitcode and IR files).
2//!
3//! All artifacts of a build land in one flat directory. A layout that mirrors the
4//! source tree instead carries the full path of every source file into the build
5//! directory, and for sources outside of the project (libraries, for example) that
6//! path goes past the limits of the Windows file system.
7
8use std::{ffi::OsStr, path::Path};
9
10use plc_util::path::path_digest;
11
12/// Upper limit for the stem in the readable part of an artifact name. The digest keeps
13/// names unique, so a longer stem has nothing to add.
14const STEM_LIMIT: usize = 32;
15
16/// Upper limit for the extension of the source in the readable part. It is cut apart
17/// from the stem, so `.st` and `.pli` units stay apart in the readable part too.
18const EXTENSION_LIMIT: usize = 16;
19
20/// Used when the unit has no file name, for example because it comes from a source
21/// container that is not a file.
22const FALLBACK_NAME: &str = "unit";
23
24/// Builds the flat artifact name for the unit that `key` identifies.
25///
26/// The name keeps the file name of the unit, which makes artifacts and linker
27/// messages readable, and adds a digest of the full `key`. The digest is what makes
28/// the name unique: two units with the same file name in different directories share
29/// one flat directory and must not overwrite each other. The full 64 bits of the
30/// digest are used because a collision would let one unit overwrite the artifact of
31/// another.
32pub fn file_name(key: &Path, extension: &str) -> String {
33    format!("{}-{:016x}.{extension}", readable_name(key), path_digest(key))
34}
35
36/// The file name of the unit with its stem and its extension each cut to their limit.
37fn readable_name(key: &Path) -> String {
38    let stem = sanitize(key.file_stem().unwrap_or_default(), STEM_LIMIT);
39    if stem.is_empty() {
40        return FALLBACK_NAME.to_string();
41    }
42
43    match key.extension() {
44        Some(extension) => format!("{stem}.{}", sanitize(extension, EXTENSION_LIMIT)),
45        None => stem,
46    }
47}
48
49/// Keeps the characters that are safe in a file name on every platform and replaces
50/// all others, so a key that holds separators or a drive letter cannot escape the
51/// artifact directory. The result holds at most `limit` characters.
52fn sanitize(name: &OsStr, limit: usize) -> String {
53    name.to_string_lossy()
54        .chars()
55        .map(|it| if it.is_ascii_alphanumeric() || matches!(it, '.' | '-' | '_') { it } else { '_' })
56        .take(limit)
57        .collect()
58}
