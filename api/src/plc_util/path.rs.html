8/// Normalize a path lexically without touching the filesystem: drop `.`
9/// components, collapse `..` against preceding normal components, preserve
10/// `..` segments that would escape an unrooted path, and keep root/prefix
11/// components. Returns `.` for the empty path.
12pub fn normalize_lexical_path(path: &Path) -> PathBuf {
13    let mut stack: Vec<Component> = Vec::new();
14
15    for component in path.components() {
16        match component {
17            Component::CurDir => {}
18            Component::ParentDir => match stack.last() {
19                Some(Component::Normal(_)) => {
20                    stack.pop();
21                }
22                Some(Component::RootDir) | Some(Component::Prefix(_)) => {
23                    // Cannot climb above the root — discard the `..`.
24                }
25                Some(Component::ParentDir) | None => {
26                    // No earlier component to cancel against, keep the `..`.
27                    stack.push(component);
28                }
29                Some(Component::CurDir) => unreachable!("CurDir is filtered out above"),
30            },
31            Component::RootDir | Component::Prefix(_) | Component::Normal(_) => {
32                stack.push(component);
33            }
34        }
35    }
36
37    let mut normalized = PathBuf::new();
38    for component in stack {
39        normalized.push(component.as_os_str());
40    }
41
42    if normalized.as_os_str().is_empty() {
43        PathBuf::from(".")
44    } else {
45        normalized
46    }
47}

49/// SipHash-1-3 digest of a path with a fixed zero key, so the result is the same on
50/// every run, process and platform.
51///
52/// The path is hashed component by component. So the same path gives the same digest no
53/// matter which separator was used to write it, while a character that is no separator
54/// on the platform, such as `\` on Linux, stays part of the name it appears in.
55pub fn path_digest(path: &Path) -> u64 {
56    let mut hasher = SipHasher13::new_with_keys(0, 0);
57    for (index, component) in path.components().enumerate() {
58        if index > 0 {
59            hasher.write(b"/");
60        }
61        hasher.write(component.as_os_str().to_string_lossy().as_bytes());
62    }
63    hasher.finish()
64}

66#[cfg(test)]
67mod tests {
68    use super::{normalize_lexical_path, path_digest};
69    use std::path::{Path, PathBuf};
70
71    #[test]
72    fn empty_path_becomes_dot() {
73        assert_eq!(normalize_lexical_path(Path::new("")), PathBuf::from("."));
74    }
75
76    #[test]
77    fn current_dir_components_are_dropped() {
78        assert_eq!(normalize_lexical_path(Path::new("./foo/./bar")), PathBuf::from("foo/bar"));
79        assert_eq!(normalize_lexical_path(Path::new(".")), PathBuf::from("."));
80    }
81
82    #[test]
83    fn parent_dir_pops_when_possible() {
84        assert_eq!(normalize_lexical_path(Path::new("foo/bar/../baz")), PathBuf::from("foo/baz"));
85        assert_eq!(normalize_lexical_path(Path::new("foo/../bar")), PathBuf::from("bar"));
86    }
87
88    #[test]
89    fn parent_dir_at_start_is_preserved() {
90        // We don't have a root to pop above, so leading `..` segments stay in the result.
91        assert_eq!(normalize_lexical_path(Path::new("../foo")), PathBuf::from("../foo"));
92        assert_eq!(normalize_lexical_path(Path::new("../../foo")), PathBuf::from("../../foo"));
93    }
94
95    #[test]
96    fn parent_dir_does_not_climb_past_root() {
97        // POSIX root: `pop()` clears the path, so `..` after root pushes literal `..`.
98        // The exact result is platform-specific but must not silently lose the leading root.
99        let normalized = normalize_lexical_path(Path::new("/foo/../bar"));
100        assert_eq!(normalized, PathBuf::from("/bar"));
101    }
102
103    #[test]
104    fn absolute_path_is_kept_absolute() {
105        assert_eq!(normalize_lexical_path(Path::new("/foo/bar")), PathBuf::from("/foo/bar"));
106        assert_eq!(normalize_lexical_path(Path::new("/foo/./bar")), PathBuf::from("/foo/bar"));
107    }
108
109    #[test]
110    fn redundant_separators_and_dots_collapse() {
111        assert_eq!(normalize_lexical_path(Path::new("foo/./././bar")), PathBuf::from("foo/bar"));
112    }
113
114    #[test]
115    fn digest_follows_the_components_and_not_the_separators() {
116        let joined = Path::new("a").join("b").join("main.st");
117
118        assert_eq!(path_digest(&joined), path_digest(Path::new("a/b/main.st")));
119        assert_ne!(path_digest(Path::new("a/b/main.st")), path_digest(Path::new("a/c/main.st")));
120    }
121}
