1/// One past the largest value a signed 64-bit integer can hold, as a float: 2^63.
2const I64_RANGE_END: f64 = (i64::MAX as u128 + 1) as f64;
3
4/// One past the largest value an unsigned 64-bit integer can hold, as a float: 2^64.
5const U64_RANGE_END: f64 = (u64::MAX as u128 + 1) as f64;
6
7/// Rounds half away from zero, then converts as the x86 `cvttsd2si` instruction does: a value
8/// outside the signed 64-bit range, NaN and the infinities included, becomes `i64::MIN`. The
9/// caller keeps the low bits of its target width, so an out-of-range value wraps and a negative
10/// value lands at the top of an unsigned range.
11fn round_to_i64(value: f64) -> i64 {
12    let rounded = value.round();
13    if (-I64_RANGE_END..I64_RANGE_END).contains(&rounded) {
14        rounded as i64
15    } else {
16        i64::MIN
17    }
18}
19
20/// The 64-bit target keeps a value in [2^63, 2^64) exact, as the x86 unsigned conversion sequence
21/// does. Everything else follows `round_to_i64`, so a negative value wraps, and NaN, the infinities
22/// and a value outside [-2^63, 2^64) become 2^63.
23fn round_to_u64(value: f64) -> u64 {
24    let rounded = value.round();
25    if (I64_RANGE_END..U64_RANGE_END).contains(&rounded) {
26        rounded as u64
27    } else {
28        round_to_i64(value) as u64
29    }
30}

32macro_rules! define_wrapping_conversion {
33    ($name:ident, $input_ty:ty, $output_ty:ty) => {
34        #[allow(non_snake_case)]
35        #[no_mangle]
36        pub extern "C" fn $name(input: $input_ty) -> $output_ty {
37            round_to_i64(f64::from(input)) as $output_ty
38        }
39    };
40}
