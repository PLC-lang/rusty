1use crate::bit_num_conversion::{whole_to_i64, whole_to_u64};

3/// One past the largest value a signed 32-bit integer can hold, as a float: 2^31.
4const I32_RANGE_END: f64 = (i32::MAX as i64 + 1) as f64;
5
6/// Converts a whole value as the x86 `cvttsd2si` instruction does with a 32-bit destination: a
7/// value outside the signed 32-bit range, NaN and the infinities included, becomes `i32::MIN`. The
8/// caller keeps the low bits of its target width, as CODESYS does for `TRUNC` and `TRUNC_INT`.
9fn whole_to_i32(whole: f64) -> i32 {
10    if (-I32_RANGE_END..I32_RANGE_END).contains(&whole) {
11        whole as i32
12    } else {
13        i32::MIN
14    }
15}

17macro_rules! define_truncating_conversion_32 {
18    ($name:ident, $input_ty:ty, $output_ty:ty) => {
19        #[allow(non_snake_case)]
20        #[no_mangle]
21        pub extern "C" fn $name(input: $input_ty) -> $output_ty {
22            whole_to_i32(f64::from(input).trunc()) as $output_ty
23        }
24    };
25}

66#[cfg(test)]
67mod tests {
68    use super::*;
69
70    #[test]
71    fn truncates_toward_zero() {
72        assert_eq!(REAL_TRUNC_INT(5.9), 5);
73        assert_eq!(REAL_TRUNC_INT(-5.9), -5);
74        assert_eq!(LREAL_TRUNC_DINT(-0.9), 0);
75        assert_eq!(LREAL_TRUNC_USINT(2.5), 2);
76    }
77
78    #[test]
79    fn negative_value_wraps_into_unsigned_target() {
80        assert_eq!(REAL_TRUNC_USINT(-5.9), 251);
81        assert_eq!(LREAL_TRUNC_UDINT(-1.9), u32::MAX);
82        assert_eq!(LREAL_TRUNC_ULINT(-1.9), u64::MAX);
83    }
84
85    #[test]
86    fn value_inside_i32_keeps_low_bits_of_target() {
87        assert_eq!(REAL_TRUNC_SINT(300.7), 44);
88        assert_eq!(LREAL_TRUNC_INT(40000.9), -25536);
89        assert_eq!(LREAL_TRUNC_UINT(2_147_483_647.9), u16::MAX);
90        assert_eq!(LREAL_TRUNC_DINT(-2_147_483_648.9), i32::MIN);
91    }
92
93    #[test]
94    fn value_outside_i32_becomes_i32_min_for_sint_to_dint() {
95        assert_eq!(LREAL_TRUNC_DINT(2_147_483_648.0), i32::MIN);
96        assert_eq!(REAL_TRUNC_INT(1.0E10), 0);
97        assert_eq!(LREAL_TRUNC_DINT(f64::NAN), i32::MIN);
98        assert_eq!(REAL_TRUNC_SINT(f32::INFINITY), 0);
99    }
100
101    #[test]
102    fn udint_is_exact_up_to_2_pow_32() {
103        assert_eq!(REAL_TRUNC_UDINT(3.0E9), 3_000_000_000);
104        assert_eq!(LREAL_TRUNC_UDINT(4_294_967_295.9), u32::MAX);
105        assert_eq!(LREAL_TRUNC_UDINT(4_294_967_296.0), 0);
106        assert_eq!(LREAL_TRUNC_UDINT(1.0E10), 1_410_065_408);
107        assert_eq!(REAL_TRUNC_UDINT(f32::NAN), 0);
108    }
109
110    #[test]
111    fn value_outside_i64_becomes_i64_min_for_udint_lint_ulint() {
112        assert_eq!(LREAL_TRUNC_LINT(1.0E10), 10_000_000_000);
113        assert_eq!(REAL_TRUNC_LINT(f32::NAN), i64::MIN);
114        assert_eq!(LREAL_TRUNC_LINT(f64::NEG_INFINITY), i64::MIN);
115        assert_eq!(REAL_TRUNC_ULINT(f32::INFINITY), 1 << 63);
116    }
117
118    #[test]
119    fn ulint_keeps_upper_half_exact() {
120        assert_eq!(LREAL_TRUNC_ULINT(1.0E19), 10_000_000_000_000_000_000);
121        assert_eq!(LREAL_TRUNC_LINT(1.0E19), i64::MIN);
122        assert_eq!(LREAL_TRUNC_ULINT(-1.0E19), 1 << 63);
123    }
124}
