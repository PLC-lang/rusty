16/// Parses an integer conversion input using the same radix rules as integer literals.
17pub fn parse_integer(input: &str) -> Option<i128> {
18    let (radix, rest) = match input.as_bytes() {
19        [b'1', b'6', b'#', ..] => (16, &input[3..]),
20        [b'0', b'x', ..] | [b'0', b'X', ..] => (16, &input[2..]),
21        [b'8', b'#', ..] => (8, &input[2..]),
22        [b'2', b'#', ..] | [b'0', b'b', ..] | [b'0', b'B', ..] => (2, &input[2..]),
23        _ => (10, input),
24    };
25    let rest = rest.replace('_', "");
26
27    if radix != 10 {
28        return i128::from_str_radix(&rest, radix).ok();
29    }
30
31    let (integer, fraction) = match rest.split_once('.') {
32        Some((integer, fraction)) if !fraction.is_empty() => (integer, fraction),
33        Some(_) => return None,
34        None => (rest.as_str(), ""),
35    };
36    if !fraction.is_empty() && !fraction.bytes().all(|byte| byte.is_ascii_digit()) {
37        return None;
38    }
39    integer.parse().ok()
40}

42/// Parses a boolean conversion input using Structured Text boolean literals and numeric forms.
43pub fn parse_bool(input: &str) -> Option<bool> {
44    match input {
45        "1" => Some(true),
46        "0" => Some(false),
47        input if input.eq_ignore_ascii_case("TRUE") => Some(true),
48        input if input.eq_ignore_ascii_case("FALSE") => Some(false),
49        _ => None,
50    }
51}

53fn parse_number<F: FromStr>(lexer: &mut ParseSession, text: &str, location: &SourceLocation) -> Option<F> {
54    match text.parse::<F>() {
55        Ok(v) => Some(v),
56        Err(_) => {
57            lexer.accept_diagnostic(
58                Diagnostic::new(format!("Failed to parse number {text}"))
59                    .with_error_code("E011")
60                    .with_location(location),
61            );
62            None
63        }
64    }
65}

67fn parse_date_from_string(
68    lexer: &mut ParseSession,
69    text: &str,
70    location: SourceLocation,
71    id: usize,
72    is_long: bool,
73) -> Option<AstNode> {
74    let mut segments = text.split('-');
75
76    //we can safely expect 3 numbers
77    let year = segments
78        .next()
79        .map(|s| parse_number::<i32>(lexer, s, &location))
80        .expect("year-segment - tokenizer broken?")?;
81    let month = segments
82        .next()
83        .map(|s| parse_number::<u32>(lexer, s, &location))
84        .expect("month-segment - tokenizer broken?")?;
85    let day = segments
86        .next()
87        .map(|s| parse_number::<u32>(lexer, s, &location))
88        .expect("day-segment - tokenizer broken?")?;
89
90    Some(AstNode::new_literal(AstLiteral::new_date_with_long_flag(year, month, day, is_long), id, location))
91}

93pub fn parse_literal_date_and_time(lexer: &mut ParseSession) -> Option<AstNode> {
94    let location = lexer.location();
95    //get rid of D# or DATE#
96    let slice = lexer.slice_and_advance().to_string();
97    let hash_location = slice.find('#').unwrap_or_default();
98    let is_long = slice[..hash_location].starts_with('L') || slice[..hash_location].starts_with('l');
99    let last_minus_location = slice.rfind('-').expect("unexpected date-and-time syntax");
100
101    let (_, date_and_time) = slice.split_at(hash_location + 1); //get rid of the prefix
102    let (date, time) = date_and_time.split_at(last_minus_location - hash_location);
103
104    //we can safely expect 3 numbers
105    let mut segments = date.split('-');
106    let msg = "unexpected date-and-time syntax";
107    let year = parse_number::<i32>(lexer, segments.next().expect(msg), &location)?;
108    let month = parse_number::<u32>(lexer, segments.next().expect(msg), &location)?;
109    let day = parse_number::<u32>(lexer, segments.next().expect(msg), &location)?;
110
111    //we can safely expect 3 numbers
112    let mut segments = time.split(':');
113    let (hour, min, sec, nano) = parse_time_of_day(lexer, &mut segments, &location)?;
114
115    let literal = if is_long {
116        AstLiteral::new_long_date_and_time(year, month, day, hour, min, sec, nano)
117    } else {
118        AstLiteral::new_date_and_time(year, month, day, hour, min, sec, nano)
119    };
120
121    Some(AstNode::new_literal(literal, lexer.next_id(), location))
122}

154fn parse_time_of_day(
155    lexer: &mut ParseSession,
156    time: &mut Split<char>,
157    location: &SourceLocation,
158) -> Option<(u32, u32, u32, u32)> {
159    let hour = parse_number::<u32>(lexer, time.next().expect("expected hours"), location)?;
160    let min = parse_number::<u32>(lexer, time.next().expect("expected minutes"), location)?;
161
162    // doesn't necessarily have to have seconds, e.g [12:00] is also valid
163    let sec = match time.next() {
164        Some(v) => parse_number::<f64>(lexer, v, location)?,
165        None => 0.0,
166    };
167
168    let nano = (sec.fract() * 1e+9_f64).round() as u32;
169
170    Some((hour, min, sec.floor() as u32, nano))
171}

173pub fn parse_literal_time(lexer: &mut ParseSession) -> Option<AstNode> {
174    const POS_D: usize = 0;
175    const POS_H: usize = 1;
176    const POS_M: usize = 2;
177    const POS_S: usize = 3;
178    const POS_MS: usize = 4;
179    const POS_US: usize = 5;
180    const POS_NS: usize = 6;
181    let location = lexer.location();
182    //get rid of T# or TIME#
183    let slice = lexer.slice_and_advance().to_string();
184    let hash_location = slice.find('#').unwrap_or_default();
185    let is_long = slice[..hash_location].starts_with('L') || slice[..hash_location].starts_with('l');
186    let (_, slice) = slice.split_at(hash_location + 1); //get rid of the prefix
187
188    let mut chars = slice.char_indices();
189    let mut char = chars.next();
190
191    let is_negative = char.map(|(_, c)| c == '-').unwrap_or(false);
192    if is_negative {
193        char = chars.next();
194    }
195
196    let mut values: [Option<f64>; 7] = [None, None, None, None, None, None, None];
197
198    let mut prev_pos = POS_D;
199    while char.is_some() {
200        //expect a number
201        let number = {
202            let start = char.expect("char").0;
203            //just eat all the digits
204            char = chars.find(|(_, ch)| !ch.is_ascii_digit() && !ch.eq(&'.'));
205            match char {
206                None => {
207                    lexer.accept_diagnostic(
208                        Diagnostic::new("Invalid TIME Literal: Cannot parse segment.")
209                            .with_error_code("E010")
210                            .with_location(location),
211                    );
212                    return None;
213                }
214                Some((index, _)) => parse_number::<f64>(lexer, &slice[start..index], &location)?,
215            }
216        };
217
218        //expect a unit
219        let unit = {
220            let start = match char {
221                Some((index, _)) => index,
222                None => {
223                    lexer.accept_diagnostic(
224                        Diagnostic::new("Invalid TIME Literal: Missing unit (d|h|m|s|ms|us|ns)")
225                            .with_error_code("E010")
226                            .with_location(location),
227                    );
228                    return None;
229                }
230            };
231
232            //just eat all the characters
233            char = chars.find(|(_, ch)| !ch.is_ascii_alphabetic());
234            &slice[start..char.unwrap_or((slice.len(), ' ')).0]
235        }
236        .to_lowercase();
237
238        //now assign the number to the according segment of the value's array
239        let position = match unit.as_str() {
240            "d" => Some(POS_D),
241            "h" => Some(POS_H),
242            "m" => Some(POS_M),
243            "s" => Some(POS_S),
244            "ms" => Some(POS_MS),
245            "us" => Some(POS_US),
246            "ns" => Some(POS_NS),
247            _ => None,
248        };
249        if let Some(position) = position {
250            //check if we assign out of order - every assignment before must have been a smaller position
251            if prev_pos > position {
252                lexer.accept_diagnostic(
253                    Diagnostic::new("Invalid TIME Literal: segments out of order, use d-h-m-s-ms")
254                        .with_error_code("E010")
255                        .with_location(location),
256                );
257                return None;
258            }
259            prev_pos = position; //remember that we wrote position
260
261            if values[position].is_some() {
262                lexer.accept_diagnostic(
263                    Diagnostic::new("Invalid TIME Literal: segments must be unique")
264                        .with_error_code("E010")
265                        .with_location(location),
266                );
267                return None;
268            }
269            values[position] = Some(number);
270        } else {
271            lexer.accept_diagnostic(
272                Diagnostic::new(format!("Invalid TIME Literal: illegal unit '{unit}'"))
273                    .with_error_code("E010")
274                    .with_location(location),
275            );
276            return None;
277        }
278    }
279
280    Some(AstNode::new_literal(
281        AstLiteral::Time(Time {
282            day: values[POS_D].unwrap_or_default(),
283            hour: values[POS_H].unwrap_or_default(),
284            min: values[POS_M].unwrap_or_default(),
285            sec: values[POS_S].unwrap_or_default(),
286            milli: values[POS_MS].unwrap_or_default(),
287            micro: values[POS_US].unwrap_or_default(),
288            nano: values[POS_NS].map(|it| it as u32).unwrap_or(0u32),
289            negative: is_negative,
290            is_long,
291        }),
292        lexer.next_id(),
293        location,
294    ))
